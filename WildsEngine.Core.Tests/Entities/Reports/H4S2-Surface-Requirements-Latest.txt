using System;
using System.Collections.Generic;
using WildsEngine.Discoveries.Abstractions;

namespace WildsEngine.Discoveries.Base
{
    /// <summary>
    /// Reusable base for projected Consequences.
    ///
    /// Consequence uses the established Discoveries identity
    /// model and remains knowledge rather than behavior.
    /// </summary>
    public abstract class ConsequenceBase :
        IdentifierBase,
        IConsequence
    {
        protected ConsequenceBase(
            string slug,
            string name,
            object? value,
            IReadOnlyList<ITruth> truths,
            double magnitude,
            double confidence,
            DateTimeOffset projectedAt)
            : base(
                slug)
        {
            Initialize(
                name,
                value,
                truths,
                magnitude,
                confidence,
                projectedAt);
        }

        protected ConsequenceBase(
            Guid id,
            string slug,
            string name,
            object? value,
            IReadOnlyList<ITruth> truths,
            double magnitude,
            double confidence,
            DateTimeOffset projectedAt)
            : base(
                id,
                slug)
        {
            Initialize(
                name,
                value,
                truths,
                magnitude,
                confidence,
                projectedAt);
        }

        public string Name { get; private set; } = null!;

        public object? Value { get; private set; }

        public IReadOnlyList<ITruth> Truths
        {
            get;
            private set;
        } = null!;

        public double Magnitude { get; private set; }

        public double Confidence { get; private set; }

        public DateTimeOffset ProjectedAt { get; private set; }

        private void Initialize(
            string name,
            object? value,
            IReadOnlyList<ITruth> truths,
            double magnitude,
            double confidence,
            DateTimeOffset projectedAt)
        {
            if (string.IsNullOrWhiteSpace(name))
            {
                throw new ArgumentException(
                    "Consequence name cannot be empty.",
                    nameof(name));
            }

            Name =
                name.Trim();

            Value =
                value;

            Truths =
                DiscoverySnapshots.Required(
                    truths,
                    nameof(truths));

            Magnitude =
                DiscoveryNumbers.NonNegative(
                    magnitude,
                    nameof(magnitude));

            Confidence =
                DiscoveryNumbers.Unit(
                    confidence,
                    nameof(confidence));

            ProjectedAt =
                projectedAt;
        }
    }
}

using System;
using System.Collections.Generic;
using WildsEngine.Discoveries.Abstractions;

namespace WildsEngine.Discoveries.Base
{
    /// <summary>
    /// Reusable immutable Discovery-vector evaluation.
    ///
    /// Magnitude is calculated by the supplied metric from the
    /// exact validated component snapshot.
    /// </summary>
    public abstract class DiscoveryVectorBase :
        IDiscoveryVector
    {
        protected DiscoveryVectorBase(
            IReadOnlyList<IDiscoveryVectorComponent> components,
            IReadOnlyList<IDiscoveryVectorContribution> contributions,
            IReadOnlyList<IDiscoveryVectorMotion> motion,
            IDiscoveryVectorMetric metric,
            DateTimeOffset evaluatedAt)
        {
            Components =
                DiscoverySnapshots.Required(
                    components,
                    nameof(components));

            Contributions =
                DiscoverySnapshots.Required(
                    contributions,
                    nameof(contributions));

            Motion =
                DiscoverySnapshots.Required(
                    motion,
                    nameof(motion));

            DiscoveryDimensions.ValidateComponents(
                Components,
                nameof(components));

            DiscoveryDimensions.ValidateContributions(
                Contributions,
                Components,
                nameof(contributions));

            DiscoveryDimensions.ValidateMotion(
                Motion,
                Components,
                nameof(motion));

            Metric =
                metric ??
                throw new ArgumentNullException(
                    nameof(metric));

            EvaluatedAt =
                evaluatedAt;

            var magnitude =
                Metric.GetMagnitude(
                    this);

            Magnitude =
                DiscoveryNumbers.NonNegative(
                    magnitude,
                    nameof(magnitude));
        }

        public IReadOnlyList<IDiscoveryVectorComponent> Components { get; }

        public int Dimension =>
            Components.Count;

        public IReadOnlyList<IDiscoveryVectorContribution> Contributions
        {
            get;
        }

        public IReadOnlyList<IDiscoveryVectorMotion> Motion { get; }

        public IDiscoveryVectorMetric Metric { get; }

        public double Magnitude { get; }

        public DateTimeOffset EvaluatedAt { get; }
    }
}

using System;
using WildsEngine.Discoveries.Abstractions;

namespace WildsEngine.Discoveries.Base
{
    public abstract class EntitySurfaceBase :
        IEntitySurface
    {
        protected EntitySurfaceBase(
            IEntity entity,
            object target,
            string alias,
            string consumerType)
        {
            Entity =
                entity ??
                throw new ArgumentNullException(
                    nameof(entity));

            Target =
                target ??
                throw new ArgumentNullException(
                    nameof(target));

            Alias =
                RequireVocabulary(
                    alias,
                    nameof(alias));

            ConsumerType =
                RequireVocabulary(
                    consumerType,
                    nameof(consumerType));
        }

        public IEntity Entity { get; }

        public object Target { get; }

        public string Alias { get; }

        public string ConsumerType { get; }

        protected void RequireOwnedFact(
            IFact fact)
        {
            if (fact == null)
            {
                throw new ArgumentNullException(
                    nameof(fact));
            }

            if (!ReferenceEquals(
                    fact.Entity,
                    Entity))
            {
                throw new ArgumentException(
                    "The Fact must belong to the exact Surface Entity.",
                    nameof(fact));
            }
        }

        protected void RequireOwnedDiscovery(
            IDiscovery discovery)
        {
            if (discovery == null)
            {
                throw new ArgumentNullException(
                    nameof(discovery));
            }

            if (!ReferenceEquals(
                    discovery.Entity,
                    Entity))

    public sealed class HypersphereDiscoveryClusterRelationshipProjector : IDiscoveryClusterRelationshipProjector
    {
        private readonly double _epsilon;
        public HypersphereDiscoveryClusterRelationshipProjector(double epsilon = 1e-10)
        {
            if (double.IsNaN(epsilon) || double.IsInfinity(epsilon) || epsilon < 0.0)
                throw new ArgumentOutOfRangeException(nameof(epsilon));
            _epsilon = epsilon;
        }

        public IDiscoveryClusterSpatialRelationship Locate(
            IDiscoveryVectorEvaluation evaluation,
            IDiscoveryClusterMapping mapping)
        {
            Validate(evaluation, mapping);
            var position = Position(evaluation.Vector, mapping);
            var delta = Delta(position, mapping.Centroid);
            var distance = Magnitude(delta);
            var signed = distance - mapping.Radius;
            var onBoundary = Math.Abs(signed) <= _epsilon;
            var nearest = new Dictionary<string,double>(StringComparer.Ordinal);
            foreach (var pair in mapping.Centroid)
            {
                var direction = distance <= _epsilon ? FirstAxis(pair.Key, mapping.Centroid) : delta[pair.Key] / distance;
                nearest[pair.Key] = pair.Value + (direction * mapping.Radius);
            }
            return new DiscoveryClusterSpatialRelationship(
                mapping, evaluation, distance, signed, signed < -_epsilon, onBoundary, nearest);
        }

        public IDiscoveryClusterTrajectoryProjection Project(
            IDiscoveryVectorMotionResult motion,
            IDiscoveryClusterMapping mapping)
        {
            if (motion == null) throw new ArgumentNullException(nameof(motion));
            var spatial = Locate(motion.Evaluation, mapping);
            var position = Position(motion.Evaluation.Vector, mapping);
            var velocity = Velocity(motion.Motion, mapping);
            var offset = Delta(position, mapping.Centroid);
            var speedSquared = Dot(velocity, velocity);
            var radialRate = spatial.DistanceToCentroid <= _epsilon ? 0.0 : Dot(offset, velocity) / spatial.DistanceToCentroid;
            if (speedSquared <= _epsilon)
                return BuildStationary(spatial, motion, radialRate);

            var closestTime = Math.Max(0.0, -Dot(offset, velocity) / speedSquared);
            var closestPosition = At(position, velocity, closestTime);
            var closestDistance = Magnitude(Delta(closestPosition, mapping.Centroid));

            var b = 2.0 * Dot(offset, velocity);
            var c = Dot(offset, offset) - (mapping.Radius * mapping.Radius);
            var discriminant = (b * b) - (4.0 * speedSquared * c);
            double? entry = null;
            double? exit = null;
            IReadOnlyDictionary<string,double>? entryPoint = null;
            IReadOnlyDictionary<string,double>? exitPoint = null;

            if (discriminant >= -_epsilon)
            {
                discriminant = Math.Max(0.0, discriminant);
                var root = Math.Sqrt(discriminant);
                var t1 = (-b - root) / (2.0 * speedSquared);
                var t2 = (-b + root) / (2.0 * speedSquared);
                if (spatial.IsInside || spatial.IsOnBoundary)
                {
                    entry = 0.0;
                    if (t2 >= 0.0) exit = t2;
                }
                else
                {
                    if (t1 >= 0.0) entry = t1;
                    if (t2 >= 0.0) exit = t2;

                if (exit.HasValue) exitPoint = At(position, velocity, exit.Value);
            }

                entry.HasValue && exit.HasValue

                spatial, motion, kind, radialRate, closestTime, closestDistance,

        }

using System.Collections.Generic;
using System.Reflection;
using WildsEngine.Discoveries.Abstractions;

namespace WildsEngine.Discoveries.Concrete
{
    public sealed class EntityNativeSurface :
        IEntityNativeSurface
    {
        private readonly IReadOnlyList<INativeAttribute> _nativeAttributes;

        public EntityNativeSurface(
            IEntity entity,
            object target)
        {
            Entity =
                entity ??
                throw new ArgumentNullException(
                    nameof(entity));

            Target =
                target ??
                throw new ArgumentNullException(
                    nameof(target));

            _nativeAttributes =
                Capture(
                    target);
        }

        public IEntity Entity { get; }

        public object Target { get; }

        public IReadOnlyList<INativeAttribute> NativeAttributes =>
            _nativeAttributes;

        private static IReadOnlyList<INativeAttribute> Capture(
            object target)
        {
            var attributes =
                new List<INativeAttribute>();

            var properties =
                target.GetType().GetProperties(
                    BindingFlags.Instance |
                    BindingFlags.Public);

            Array.Sort(
                properties,
                (left, right) =>
                    string.CompareOrdinal(
                        left.Name,
                        right.Name));

            for (var index = 0;
                 index < properties.Length;
                 index++)
            {
                var property =
                    properties[index];

                if (property.GetIndexParameters().Length != 0)
                {
                    continue;
                }

                var canRead =
                    property.GetMethod != null &&
                    property.GetMethod.IsPublic;

                if (!canRead)
                {
                    continue;
                }

                var canWrite =
                    property.SetMethod != null &&
                    property.SetMethod.IsPublic;

                attributes.Add(
                    new NativeAttribute(
                        property.Name,
                        property.PropertyType,
                        property.GetValue(target),
                        canRead,
                        canWrite));
            }

            var fields =
                target.GetType().GetFields(
                    BindingFlags.Instance |
                    BindingFlags.Public);

            Array.Sort(
                fields,
                (left, right) =>
                    string.CompareOrdinal(
                        left.Name,
                        right.Name));

            for (var index = 0;
                 index < fields.Length;
                 index++)
            {
                var field =
                    fields[index];

                attributes.Add(
                    new NativeAttribute(
                        field.Name,
                        field.FieldType,
                        field.GetValue(target),
                        true,
                        !field.IsInitOnly && !field.IsLiteral));
            }

            return attributes;
        }
    }
}

using System;
using System.Collections.Generic;
using System.Collections.ObjectModel;
using WildsEngine.Discoveries.Abstractions;
using WildsEngine.Discoveries.Abstractions.Motion;

namespace WildsEngine.Discoveries.Motion
{
    public sealed class DiscoveryVectorEvaluationSeries : IDiscoveryVectorEvaluationSeries
    {
        public DiscoveryVectorEvaluationSeries(
            string scope,
            IReadOnlyList<IDiscoveryVectorInput> inputs,
            IReadOnlyList<IDiscoveryVectorEvaluation> evaluations)
        {
            if (string.IsNullOrWhiteSpace(scope)) throw new ArgumentException("Series requires a scope.", nameof(scope));
            if (inputs == null) throw new ArgumentNullException(nameof(inputs));
            if (evaluations == null) throw new ArgumentNullException(nameof(evaluations));
            if (inputs.Count == 0) throw new ArgumentException("Series requires at least one input.", nameof(inputs));
            if (evaluations.Count == 0) throw new ArgumentException("Series requires at least one evaluation.", nameof(evaluations));
            var inputCopy = new IDiscoveryVectorInput[inputs.Count];
            for (var i = 0; i < inputs.Count; i++) inputCopy[i] = inputs[i] ?? throw new ArgumentException("Inputs cannot contain null.", nameof(inputs));
            var evalCopy = new IDiscoveryVectorEvaluation[evaluations.Count];
            DateTimeOffset? prior = null;
            for (var i = 0; i < evaluations.Count; i++)
            {
                var evaluation = evaluations[i] ?? throw new ArgumentException("Evaluations cannot contain null.", nameof(evaluations));
                if (prior.HasValue && evaluation.EvaluatedAt <= prior.Value)
                    throw new ArgumentException("Evaluation timestamps must be strictly increasing.", nameof(evaluations));
                ValidateInputs(inputCopy, evaluation);
                evalCopy[i] = evaluation; prior = evaluation.EvaluatedAt;
            }
            Scope = scope.Trim();
            Inputs = new ReadOnlyCollection<IDiscoveryVectorInput>(inputCopy);
            Evaluations = new ReadOnlyCollection<IDiscoveryVectorEvaluation>(evalCopy);
        }
        public string Scope { get; }
        public IReadOnlyList<IDiscoveryVectorInput> Inputs { get; }
        public IReadOnlyList<IDiscoveryVectorEvaluation> Evaluations { get; }
        public IDiscoveryVectorEvaluation? Current => Evaluations.Count == 0 ? null : Evaluations[Evaluations.Count - 1];
        public IDiscoveryVectorEvaluation? Previous => Evaluations.Count < 2 ? null : Evaluations[Evaluations.Count - 2];

        private static void ValidateInputs(
            IReadOnlyList<IDiscoveryVectorInput> inputs,
            IDiscoveryVectorEvaluation evaluation)
        {
            for (var c = 0; c < evaluation.Vector.Contributions.Count; c++)
            {
                var candidate = evaluation.Vector.Contributions[c].Input;
                var found = false;
                for (var i = 0; i < inputs.Count; i++)
                    if (ReferenceEquals(candidate, inputs[i])) { found = true; break; }
                if (!found) throw new ArgumentException("Evaluation contains a contribution from outside the series input population.");
            }
        }
    }
}

using System.Collections.Generic;

    public sealed class DiscoveryRecognizedGeometryPopulation :
        IDiscoveryRecognizedGeometryPopulation
    {
        public DiscoveryRecognizedGeometryPopulation(
            Guid id,
            IDiscoveryGeometryRegistration registration,
            IReadOnlyList<IDiscoveryClusterRecognitionResult> members,
            DateTimeOffset createdAt)
        {
            if (id == Guid.Empty)
                throw new ArgumentException("Population identity is required.", nameof(id));
            Registration = registration ?? throw new ArgumentNullException(nameof(registration));
            if (members == null || members.Count == 0)
                throw new ArgumentException("At least one recognized geometry result is required.", nameof(members));

            var allowedMappings = new HashSet<Guid>(registration.MappingIds);
            var memberCopy = new IDiscoveryClusterRecognitionResult[members.Count];
            var semanticKeys = new List<string>();
            var seenSemanticKeys = new HashSet<string>(StringComparer.Ordinal);

            for (var index = 0; index < members.Count; index++)
            {
                var member = members[index] ??
                    throw new ArgumentException("Population members cannot contain null.", nameof(members));
                if (!member.IsRecognized)
                    throw new ArgumentException("Every population member must be recognized.", nameof(members));
                if (!allowedMappings.Contains(member.Mapping.Id))
                    throw new ArgumentException("Every population member must belong to the supplied geometry registration.", nameof(members));
                memberCopy[index] = member;
                if (seenSemanticKeys.Add(member.Mapping.SemanticKey))
                    semanticKeys.Add(member.Mapping.SemanticKey);
            }

            semanticKeys.Sort(StringComparer.Ordinal);
            Id = id;
            Scope = registration.Scope;
            Members = new ReadOnlyCollection<IDiscoveryClusterRecognitionResult>(memberCopy);
            SemanticKeys = new ReadOnlyCollection<string>(semanticKeys);
            CreatedAt = createdAt;
        }

        public Guid Id { get; }
        public string Scope { get; }
        public IDiscoveryGeometryRegistration Registration { get; }
        public IReadOnlyList<IDiscoveryClusterRecognitionResult> Members { get; }
        public IReadOnlyList<string> SemanticKeys { get; }
        public DateTimeOffset CreatedAt { get; }
    }

using System.Collections.Generic;
using System.Collections.ObjectModel;
using WildsEngine.Discoveries.Abstractions;

namespace WildsEngine.Discoveries.Policies
{
    /// <summary>
    /// Counts Evidence only when every supplied policy accepts it.
    /// </summary>
    public sealed class AllEvidencePolicies :
        EvidenceContributionPolicyBase
    {
        private readonly IReadOnlyList<IEvidenceContributionPolicy> _policies;

        public AllEvidencePolicies(
            IReadOnlyList<IEvidenceContributionPolicy> policies)
        {
            if (policies == null)
            {
                throw new ArgumentNullException(
                    nameof(policies));
            }

            if (policies.Count == 0)
            {
                throw new ArgumentException(
                    "At least one policy is required.",
                    nameof(policies));
            }

            var snapshot =
                new List<IEvidenceContributionPolicy>(
                    policies.Count);

            for (var index = 0;
                 index < policies.Count;
                 index++)
            {
                snapshot.Add(
                    policies[index] ??
                    throw new ArgumentException(
                        "Policies cannot contain null.",
                        nameof(policies)));
            }

            _policies =
                new ReadOnlyCollection<IEvidenceContributionPolicy>(
                    snapshot);
        }

        public IReadOnlyList<IEvidenceContributionPolicy> Policies =>
            _policies;

        protected override bool Evaluate(
            IDiscoveryEvidence evidence)
        {
            for (var index = 0;
                 index < _policies.Count;
                 index++)
            {
                if (!_policies[index].Contributes(evidence))
                {
                    return false;
                }
            }

            return true;
        }
    }
}

using System;
using System.Collections.Generic;
using WildsEngine.Discoveries.Abstractions.Clusters;
using WildsEngine.Discoveries.Base;

namespace WildsEngine.Discoveries.Policies
{
    public sealed class BoundaryVarianceClusterStabilityMetric :
        IdentifierBase,
        IDiscoveryClusterStabilityMetric
    {
        public BoundaryVarianceClusterStabilityMetric(
            string slug = "boundary-variance")
            : base(
                Guid.NewGuid(),
                slug)
        {
        }

        public double GetStability(
            IReadOnlyList<IDiscoveryClusterObservation> observations)
        {
            if (observations == null ||
                observations.Count == 0)
            {
                throw new ArgumentException(
                    "At least one observation is required.",
                    nameof(observations));
            }

            if (observations.Count == 1)
            {
                return 1.0;
            }

            double mean = 0.0;

            for (var index = 0;
                 index < observations.Count;
                 index++)
            {
                mean +=
                    observations[index]
                        .Relationship
                        .SignedDistanceToBoundary;
            }

            mean /= observations.Count;

            double variance = 0.0;

            for (var index = 0;
                 index < observations.Count;
                 index++)
            {
                var delta =
                    observations[index]
                        .Relationship
                        .SignedDistanceToBoundary -
                    mean;

                variance +=
                    delta * delta;
            }

            variance /=
                observations.Count;

            return
                1.0 /
                (1.0 + variance);
        }
    }
}

using System;
using WildsEngine.Discoveries.Abstractions;

namespace WildsEngine.Discoveries.Policies
{
    /// <summary>
    /// Provides null validation for reusable Evidence contribution policies.
    /// </summary>
    public abstract class EvidenceContributionPolicyBase :
        IEvidenceContributionPolicy
    {
        public bool Contributes(
            IDiscoveryEvidence evidence)
        {
            if (evidence == null)
            {
                throw new ArgumentNullException(
                    nameof(evidence));
            }

            return Evaluate(
                evidence);
        }

        protected abstract bool Evaluate(
            IDiscoveryEvidence evidence);

        protected static double RequireNormalized(
            double value,
            string parameterName)
        {
            if (double.IsNaN(value) ||
                double.IsInfinity(value) ||
                value < 0.0 ||
                value > 1.0)
            {
                throw new ArgumentOutOfRangeException(
                    parameterName,
                    "The value must be between zero and one.");
            }

            return value;
        }
    }
}

        private static IReadOnlyList<IDiscoveryClusterObservation>
            GetCurrentOccupancyObservations(
                IReadOnlyList<IDiscoveryClusterObservation> observations)
        {
            if (observations == null)
            {
                throw new ArgumentNullException(
                    nameof(observations));
            }

            if (observations.Count == 0)
            {
                throw new ArgumentException(
                    "At least one observation is required.",
                    nameof(observations));
            }

            var current =
                observations[observations.Count - 1];

            var currentIsInside =
                current.Relationship.IsInside ||
                current.Relationship.IsOnBoundary;

            // There is no active occupancy interval.
            //
            // Preserve the current observation for the metric rather
            // than treating previous occupancy as current stability.
            if (!currentIsInside)
            {
                return new[]
                {
                    current
                };
            }

            var startIndex =
                observations.Count - 1;

            while (startIndex > 0)
            {
                var previous =
                    observations[startIndex - 1];

                var previousIsInside =
                    previous.Relationship.IsInside ||
                    previous.Relationship.IsOnBoundary;

                if (!previousIsInside)
                {
                    break;
                }

                startIndex--;
            }

            var result =
                new List<IDiscoveryClusterObservation>(
                    observations.Count - startIndex);

            for (var index = startIndex;
                 index < observations.Count;
                 index++)
            {
                result.Add(
                    observations[index]);
            }

            return result.AsReadOnly();
        }
    }

        public DiscoveryVectorizer(

        {

            _metric = metric ?? throw new ArgumentNullException(nameof(metric));
        }

        public async Task<IDiscoveryVectorizationResult> VectorizeAsync(
            IDiscoveryVectorInput input,
            object? provenance = null,
            Guid? parentExecutionId = null,
            Guid? pulseId = null,
            CancellationToken cancellationToken = default)
        {
            if (input == null) throw new ArgumentNullException(nameof(input));
            var serviceResult = await _service.ExecuteAsync(
                new DiscoveryVectorizationRequest(input),
                provenance,
                parentExecutionId,
                pulseId,
                cancellationToken);

            var contributions = serviceResult.Value.Contributions;
            ValidateExactInput(input, contributions);
            var components = Aggregate(contributions);
            var vector = new DiscoveryVector(
                components,
                contributions,
                new List<IDiscoveryVectorMotion>(),
                _metric,
                DateTimeOffset.UtcNow);
            return new DiscoveryVectorizationResult(serviceResult.ExecutionId, vector);
        }

        private static void ValidateExactInput(
            IDiscoveryVectorInput input,
            IReadOnlyList<IDiscoveryVectorContribution> contributions)
        {
            if (contributions.Count == 0)
                throw new InvalidOperationException("Vectorization service returned no contributions.");
            for (var index = 0; index < contributions.Count; index++)
                if (!ReferenceEquals(input, contributions[index].Input))
                    throw new InvalidOperationException(
                        "Every vectorization contribution must reference the exact requested input.");
        }

        private static IReadOnlyList<IDiscoveryVectorComponent> Aggregate(
            IReadOnlyList<IDiscoveryVectorContribution> contributions)
        {
            var totals = new SortedDictionary<string, double>(StringComparer.Ordinal);
            for (var index = 0; index < contributions.Count; index++)
            {
                var contribution = contributions[index];
                if (string.IsNullOrWhiteSpace(contribution.Dimension))
                    throw new ArgumentException("Contribution dimension cannot be empty.", nameof(contributions));
                if (double.IsNaN(contribution.Value) || double.IsInfinity(contribution.Value))
                    throw new ArgumentOutOfRangeException(nameof(contributions), "Contribution value must be finite.");
                if (double.IsNaN(contribution.Weight) || double.IsInfinity(contribution.Weight) || contribution.Weight < 0.0)
                    throw new ArgumentOutOfRangeException(nameof(contributions), "Contribution weight must be finite and non-negative.");
                var weighted = contribution.Value * contribution.Weight;
                if (double.IsNaN(weighted) || double.IsInfinity(weighted))
                    throw new ArgumentOutOfRangeException(nameof(contributions), "Weighted contribution must be finite.");
                totals.TryGetValue(contribution.Dimension, out var current);
                var next = current + weighted;
                if (double.IsNaN(next) || double.IsInfinity(next))
                    throw new ArgumentOutOfRangeException(nameof(contributions), "Aggregated component must be finite.");
                totals[contribution.Dimension] = next;
            }
            var components = new List<IDiscoveryVectorComponent>(totals.Count);
            foreach (var pair in totals)
                components.Add(new DiscoveryVectorComponent(pair.Key, pair.Value));
            return components;
        }
    }
