        [Fact]
        public void PulseCannotResolveUntilAllRequiredWorkCompletes()
        {
            TestLog.Section(
                "Pulse Coordination Certification");

            TestLog.Test(
                "Pulse waits for every required WorkUnit");

            var request =
                new TestRuntimeObject(
                    "player-request");

            var pulse =
                new Pulse(
                    request);

            Assert.True(
                pulse.TryBeginDecomposition());

            var first =
                CreateWork(
                    pulse,
                    "QSL");

            var second =
                CreateWork(
                    pulse,
                    "Instruction");

            var third =
                CreateWork(
                    pulse,
                    "Task");

            var coordinator =
                new PulseCoordinator(
                    pulse);

            Assert.True(
                coordinator.AddWorkUnit(
                    first));

            Assert.True(
                coordinator.AddWorkUnit(
                    second));

            Assert.True(
                coordinator.AddWorkUnit(
                    third));

            TestLog.State(
                $"Pulse {pulse.Id} owns " +
                $"{coordinator.WorkUnitCount} required WorkUnits");

            Assert.True(
                pulse.TryMarkPending());

            Assert.True(
                pulse.TryMarkReady());

            Assert.True(
                pulse.TryBeginProcessing());

            BeginAndComplete(
                first,
                "qsl-result");

            TestLog.Task(
                "WorkUnit A completed");

            BeginAndComplete(
                second,
                "instruction-result");

            TestLog.Task(
                "WorkUnit B completed");

            Assert.True(
                third.TryBeginProcessing());

            TestLog.Task(
                "WorkUnit C still processing");

            Assert.False(
                coordinator.CanResolve);

            Assert.False(
                coordinator.TryResolve(
                    "too-early",
                    out var earlyMoment));

            Assert.Null(
                earlyMoment);

            Assert.Equal(
                PulseStatus.Processing,
                pulse.Status);

            TestLog.Expected(
                "Pulse refused early completion");

            var finalResult =
                new TestRuntimeObject(
                    "final-result");

            Assert.True(
                third.TryComplete(
                    finalResult));

            TestLog.Task(
                "WorkUnit C completed");

            Assert.True(
                coordinator.CanResolve);

            Assert.True(
                coordinator.TryResolve(
                    finalResult,
                    out var moment));

            Assert.NotNull(
                moment);

            Assert.Equal(
                PulseStatus.Completed,
                pulse.Status);

            Assert.True(
                pulse.IsTerminal);

            Assert.Equal(
                pulse.Id,
                moment!.PulseId);

            Assert.Equal(
                3,
                moment.WorkUnitIds.Count);

            Assert.Contains(
                first.Id,
                moment.WorkUnitIds);

            Assert.Contains(
                second.Id,
                moment.WorkUnitIds);

            Assert.Contains(
                third.Id,
                moment.WorkUnitIds);

            Assert.Same(
                finalResult,
                moment.Result);

            TestLog.State(
                "All required WorkUnits resolved");

            TestLog.State(
                $"Moment {moment.Id} closed Pulse {pulse.Id}");

            TestLog.Pass(
                "Pulse completed only after all required work completed");
        }

using System;

    /// <summary>
    /// Certifies the QSL -> WildsEngine integration boundary.
    ///
    /// QSL source
    ///     -> QslEngine
    ///     -> QslProcessingResult
    ///     -> WildsEngine
    ///     -> Pulse
    ///     -> WorkUnit
    ///     -> Moment
    ///     -> Timeline
    ///     -> Epoch
    ///
    /// These tests verify that QSL-produced runtime values
    /// survive the complete Wilds processing route without
    /// losing their identity or outcome semantics.
    /// </summary>
    public sealed class QslWildsIntegrationTests
    {
        [Fact]
        public async Task
            CertifiedQslArithmeticTraversesCompleteWildsRoute()
        {
            TestLog.Section(
                "QSL -> WildsEngine Integration");

            TestLog.Test(
                "Certified QSL arithmetic traverses complete Wilds route");

            var qslEngine =
                CreateQslEngine();

            const string source =
                "1 + 2";

            TestLog.Write(
                $"[QSL] Evaluating source: {source}");

            var qslResult =
                qslEngine.Evaluate(
                    source);

            Assert.True(
                qslResult.Succeeded);

            Assert.Empty(
                qslResult.ParserDiagnostics);

            Assert.Empty(
                qslResult.RuntimeDiagnostics);

            Assert.Null(
                qslResult.Exception);

            TestLog.State(
                $"QSL evaluation succeeded | " +
                $"Value={qslResult.Value} | " +
                $"Type={qslResult.Value?.GetType().FullName ?? "null"}");

            var component =
                CreateComponent(
                    "qsl.integration.arithmetic");

            using var wildsEngine =
                CreateWildsEngine(
                    component);

            Assert.True(
                wildsEngine.Initialize());

            TestLog.State(
                "WildsEngine initialized");

            var moment =
                await wildsEngine.ExecuteAsync(
                    qslResult,
                    (
                        request,
                        cancellationToken) =>
                    {
                        cancellationToken
                            .ThrowIfCancellationRequested();

                        Assert.Same(
                            qslResult,
                            request);

                        TestLog.State(
                            "Exact QslProcessingResult reached " +
                            "Wilds execution");

                        return new ValueTask<object?>(
                            qslResult.Value);
                    });

            TestLog.State(
                $"Wilds Moment resolved | " +
                $"Outcome={moment.Outcome} | " +
                $"PulseId={moment.PulseId} | " +
                $"WorkUnits={moment.WorkUnitIds.Count}");

            TestLog.State(
                $"Wilds result | " +
                $"Value={moment.Result} | " +
                $"Type={moment.Result?.GetType().FullName ?? "null"}");

            Assert.Equal(
                MomentOutcome.Completed,
                moment.Outcome);

            Assert.Null(
                moment.Exception);

            Assert.Single(
                moment.WorkUnitIds);

            /*
             * This is intentionally stronger than:
             *
             * Assert.Equal(3, moment.Result);
             *
             * The integration boundary is responsible for
             * preserving the runtime value produced by QSL.
             * It is not responsible for coercing that value
             * into a test-selected numeric type.
             */
            Assert.Same(
                qslResult.Value,
                moment.Result);

            Assert.Equal(
                qslResult.Value?.GetType(),
                moment.Result?.GetType());

            Assert.Single(
                wildsEngine.Timeline.Moments);

            Assert.Same(
                moment,
                wildsEngine.Timeline.Moments[0]);

            Assert.Single(
                wildsEngine.Epoch.Timelines);

            Assert.Same(
                wildsEngine.Timeline,
                wildsEngine.Epoch.Timelines[0]);

            Assert.Same(
                qslResult.Value,
                wildsEngine.Timeline
                    .Moments[0]
                    .Result);

            TestLog.State(
                "Exact QSL-produced value survived " +
                "Pulse -> WorkUnit -> Moment -> " +
                "Timeline -> Epoch");

            TestLog.Pass(
                "Certified QSL arithmetic traversed " +
                "complete Wilds route");
        }

            ConcurrentLoadRespectsLimit()
        {
            const int limit = 3;
            const int count = 24;

            using var engine =
                CreateEngine(
                    "certification.concurrent",
                    retainedInstructions: limit,
                    concurrentInstructions: limit);

            Assert(
                engine.Initialize(),
                "Initialize() should succeed.");

            var current = 0;
            var maximum = 0;

            var release =
                new TaskCompletionSource<bool>(
                    TaskCreationOptions
                        .RunContinuationsAsynchronously);

            var entered =
                new TaskCompletionSource<bool>(
                    TaskCreationOptions
                        .RunContinuationsAsynchronously);

            var tasks =
                new List<Task<int>>(
                    count);

            for (var i = 0; i < count; i++)
            {
                tasks.Add(
                    engine.ExecuteInstructionAsync<int>(
                            async (_, token) =>
                            {
                                var now =
                                    Interlocked.Increment(
                                        ref current);

                                UpdateMaximum(
                                    ref maximum,
                                    now);

                                if (now >= limit)
                                {
                                    entered.TrySetResult(
                                        true);
                                }

                                try
                                {
                                    await release.Task
                                        .ConfigureAwait(false);

                                    token.ThrowIfCancellationRequested();

                                    return 1;
                                }
                                finally
                                {
                                    Interlocked.Decrement(
                                        ref current);
                                }
                            })
                        .AsTask());
            }

            await entered.Task
                .ConfigureAwait(false);

            LogPool(
                engine,
                "At concurrency limit");

            Assert(
                engine.Instructions.ActiveCount <= limit,
                "Active leases exceeded MaximumConcurrent.");

            release.TrySetResult(
                true);

            await Task.WhenAll(
                    tasks)
                .ConfigureAwait(false);

            Assert(
                maximum == limit,
                "Concurrent load did not exercise the configured limit.");

            Assert(
                engine.Instructions.ActiveCount == 0,
                "All concurrent leases should be returned.");

            Assert(
                engine.Instructions.CreatedCount <= limit,
                "Pool created more processors than " +
                "concurrent leasing permits.");
        }

        private static void UpdateMaximum(
            ref int target,
            int value)
        {
            while (true)
            {
                var current =
                    Volatile.Read(
                        ref target);

                if (value <= current)
                    return;

                if (Interlocked.CompareExchange(
                        ref target,
                        value,
                        current) == current)
                {
                    return;
                }
            }
        }

using System.Collections.Generic;

namespace WildsEngine.Core.Abstractions
{
    /// <summary>
    /// Coordinates the internal work required to resolve
    /// one complete Pulse.
    /// </summary>
    public interface IPulseCoordinator
    {
        IPulse Pulse { get; }

        int WorkUnitCount { get; }

        IReadOnlyList<IWorkUnit> WorkUnits { get; }

        bool AddWorkUnit(
            IWorkUnit workUnit);

        /// <summary>
        /// True only when every required WorkUnit completed
        /// successfully.
        /// </summary>
        bool CanResolve { get; }

        /// <summary>
        /// True when every required WorkUnit has reached any
        /// terminal state.
        /// </summary>
        bool CanFinalize { get; }

        /// <summary>
        /// Resolves a successful Pulse.
        /// </summary>
        bool TryResolve(
            object? result,
            out IMoment? moment);

        /// <summary>
        /// Finalizes a Pulse according to the terminal state
        /// of all required WorkUnits.
        /// </summary>
        bool TryFinalize(
            object? result,
            out IMoment? moment);
    }
}

using System;
using System.Threading;
using System.Threading.Tasks;
using WildsEngine.Core.Abstractions;
using WildsEngine.Core.Pooling;
using WildsEngine.Core.Processing.Coordinator;
using WildsEngine.Core.Processing.Processor;
using WildsEngine.Core.Processing.Registry;
using WildsEngine.Core.Units;

namespace WildsEngine.Core
{
    /// <summary>
    /// Composition root for one isolated Wilds runtime instance.
    ///
    /// Wilds accepts complete requests as Pulses, owns their
    /// internal WorkUnits, resolves them into Moments, and
    /// records those Moments in its Timeline and Epoch.
    /// </summary>
    public sealed class WildsEngine :
        IDisposable
    {
        private readonly IComponent _component;

        private readonly DefaultInstructionsRegistry
            _instructionRegistry;

        private readonly DefaultTasksRegistry
            _tasksRegistry;

        private readonly ObjectPool<ITaskProcessor>
            _taskProcessors;

        private readonly ObjectPool<IInstructionsProcessor>
            _instructionsProcessors;

        private readonly Timeline _timeline;
        private readonly Epoch _epoch;

        private int _disposed;

        public WildsEngine(
            IComponent component,
            WildsEngineOptions options)
        {
            _component = component ??
                throw new ArgumentNullException(
                    nameof(component));

            if (options == null)
            {
                throw new ArgumentNullException(
                    nameof(options));
            }

            if (string.IsNullOrWhiteSpace(
                    options.Layer))
            {
                throw new ArgumentException(
                    "The engine requires a layer.",
                    nameof(options));
            }

            Layer =
                options.Layer.Trim();

            _instructionRegistry =
                new DefaultInstructionsRegistry(
                    _component,
                    $"{Layer}.instruction-registry");

            _tasksRegistry =
                new DefaultTasksRegistry(
                    _component,
                    $"{Layer}.task-registry");

            _taskProcessors =
                new ObjectPool<ITaskProcessor>(
                    factory: () =>
                        new TaskProcessor(
                            _component,
                            $"{Layer}.task-processor",
                            _tasksRegistry),
                    maximumRetained:
                        options.RetainedTaskProcessors,
                    maximumConcurrent:
                        options.ConcurrentTaskProcessors);

            _instructionsProcessors =
                new ObjectPool<IInstructionsProcessor>(
                    factory: () =>
                        new InstructionsProcessor(
                            _component,
                            $"{Layer}.instructions-processor",
                            _instructionRegistry,
                            _taskProcessors),
                    maximumRetained:
                        options.RetainedInstructionsProcessors,
                    maximumConcurrent:
                        options.ConcurrentInstructionsProcessors);

            _timeline =
                new Timeline();

            _epoch =
                new Epoch();

            if (!_epoch.AddTimeline(
                    _timeline))
            {
                throw new InvalidOperationException(
                    "The engine could not initialize " +
                    "its primary Timeline.");
            }
        }

        public string Layer { get; }

        public IComponent Component =>
            _component;

        public bool IsInitialized { get; private set; }

        public IObjectPool<IInstructionsProcessor>
            Instructions =>
                _instructionsProcessors;

        /// <summary>
        /// Contains the ordered Moment history produced by
        /// this engine instance.
        /// </summary>
        public ITimeline Timeline =>
            _timeline;

        /// <summary>
        /// Contains the Timeline history owned by this
        /// engine instance.
        /// </summary>
        public IEpoch Epoch =>
            _epoch;

        public bool Initialize()
        {
            ThrowIfDisposed();

            if (IsInitialized)
                return false;

            _instructionRegistry.Initialize();
            _tasksRegistry.Initialize();

            IsInitialized = true;

            return true;
        }

        /// <summary>
        /// Executes one complete consumer request through the
        /// certified Pulse lifecycle.
        ///
        /// Wilds creates and correlates the Pulse and WorkUnit,
        /// owns their lifecycle transitions, finalizes the
        /// terminal Moment, and records that Moment in the
        /// engine Timeline.
        ///
        /// The supplied operation represents internal work
        /// performed by Wilds or an integration adapter.
        /// </summary>
        public async ValueTask<IMoment> ExecuteAsync(
            object? request,
            Func<
                object?,
                CancellationToken,
                ValueTask<object?>> operation,
            CancellationToken cancellationToken = default)
        {
            ThrowIfDisposed();
            ThrowIfNotInitialized();

            if (operation == null)
            {
                throw new ArgumentNullException(
                    nameof(operation));
            }

            var pulse =
                new Pulse(
                    request);

            var work =
                new WorkUnit(
                    pulse.Id,
                    request);

            var coordinator =
                new PulseCoordinator(
                    pulse);

            BeginPulse(
                pulse,
                work,
                coordinator);

            object? result;

            try
            {
                cancellationToken
                    .ThrowIfCancellationRequested();

                result =
                    await operation(
                            request,
                            cancellationToken)
                        .ConfigureAwait(false);

                cancellationToken
                    .ThrowIfCancellationRequested();

                if (!work.TryComplete(
                        result))
                {
                    throw new InvalidOperationException(
                        "The WorkUnit could not complete.");
                }
            }
            catch (OperationCanceledException)
            {
                if (!work.IsTerminal &&
                    !work.TryCancel())
                {
                    throw new InvalidOperationException(
                        "The WorkUnit could not be cancelled.");
                }

                var cancelledMoment =
                    FinalizePulse(
                        coordinator,
                        null);

                RecordMoment(
                    cancelledMoment);

                return cancelledMoment;
            }
            catch (Exception exception)
            {
                if (!work.IsTerminal &&
                    !work.TryFail(
                        exception))
                {
                    throw new InvalidOperationException(
                        "The WorkUnit could not record failure.",
                        exception);
                }

                var failedMoment =
                    FinalizePulse(
                        coordinator,
                        null);

                RecordMoment(
                    failedMoment);

                return failedMoment;
            }

            var completedMoment =
                FinalizePulse(
                    coordinator,
                    result);

            RecordMoment(
                completedMoment);

            return completedMoment;
        }

        /// <summary>
        /// Preserves the existing certified instruction
        /// processor lease boundary.
        /// </summary>
        public async ValueTask<TResult>
            ExecuteInstructionAsync<TResult>(
                Func<
                    IInstructionsProcessor,
                    CancellationToken,
                    ValueTask<TResult>> operation,
                CancellationToken cancellationToken = default)
        {
            ThrowIfDisposed();
            ThrowIfNotInitialized();

            if (operation == null)
            {
                throw new ArgumentNullException(
                    nameof(operation));
            }

            using (var lease =
                await _instructionsProcessors
                    .RentAsync(
                        cancellationToken)
                    .ConfigureAwait(false))
            {
                return await operation(
                        lease.Instance,
                        cancellationToken)
                    .ConfigureAwait(false);
            }
        }

        public bool Shutdown()
        {
            ThrowIfDisposed();

            if (!IsInitialized)
                return false;

            _tasksRegistry.Shutdown();
            _instructionRegistry.Shutdown();

            IsInitialized = false;

            return true;
        }

        public void Dispose()
        {
            if (Interlocked.Exchange(
                    ref _disposed,
                    1) != 0)
            {
                return;
            }

            if (IsInitialized)
            {
                _tasksRegistry.Shutdown();
                _instructionRegistry.Shutdown();

                IsInitialized = false;
            }

            _instructionsProcessors.Dispose();
            _taskProcessors.Dispose();
        }

        private static void BeginPulse(
            Pulse pulse,
            WorkUnit work,
            IPulseCoordinator coordinator)
        {
            if (!pulse.TryBeginDecomposition())
            {
                throw new InvalidOperationException(
                    "The Pulse could not begin decomposition.");
            }

            if (!coordinator.AddWorkUnit(
                    work))
            {
                throw new InvalidOperationException(
                    "The WorkUnit could not be correlated " +
                    "with its Pulse.");
            }

            if (!work.TryMarkPending())
            {
                throw new InvalidOperationException(
                    "The WorkUnit could not become pending.");
            }

            if (!pulse.TryMarkPending())
            {
                throw new InvalidOperationException(
                    "The Pulse could not become pending.");
            }

            if (!pulse.TryMarkReady())
            {
                throw new InvalidOperationException(
                    "The Pulse could not become ready.");
            }

            if (!pulse.TryBeginProcessing())
            {
                throw new InvalidOperationException(
                    "The Pulse could not begin processing.");
            }

            if (!work.TryBeginProcessing())
            {
                throw new InvalidOperationException(
                    "The WorkUnit could not begin processing.");
            }
        }

        private static IMoment FinalizePulse(
            IPulseCoordinator coordinator,
            object? result)
        {
            if (!coordinator.CanFinalize)
            {
                throw new InvalidOperationException(
                    "The Pulse cannot finalize while required " +
                    "WorkUnits remain unresolved.");
            }

            if (!coordinator.TryFinalize(
                    result,
                    out var moment))
            {
                throw new InvalidOperationException(
                    "The Pulse coordinator could not create " +
                    "a terminal Moment.");
            }

            return moment ??
                throw new InvalidOperationException(
                    "Pulse finalization returned no Moment.");
        }

        private void RecordMoment(
            IMoment moment)
        {
            if (!_timeline.AddMoment(
                    moment))
            {
                throw new InvalidOperationException(
                    "The terminal Moment could not be " +
                    "recorded in the engine Timeline.");
            }
        }

        private void ThrowIfNotInitialized()
        {
            if (!IsInitialized)
            {
                throw new InvalidOperationException(
                    "The WildsEngine must be initialized " +
                    "before processing requests.");
            }
        }

        private void ThrowIfDisposed()
        {
            if (Volatile.Read(
                    ref _disposed) != 0)
            {
                throw new ObjectDisposedException(
                    nameof(WildsEngine));
            }
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using WildsEngine.Core.Abstractions;
using WildsEngine.Core.Units;

namespace WildsEngine.Core.Processing.Coordinator
{
    /// <summary>
    /// Coordinates all required WorkUnits for one Pulse.
    ///
    /// The coordinator prevents early completion and records
    /// successful, failed, and cancelled Pulse outcomes.
    /// </summary>
    public sealed class PulseCoordinator :
        IPulseCoordinator
    {
        private readonly object _sync =
            new object();

        private readonly List<IWorkUnit> _workUnits =
            new List<IWorkUnit>();

        private readonly HashSet<Guid> _workUnitIds =
            new HashSet<Guid>();

        public PulseCoordinator(
            IPulse pulse)
        {
            Pulse = pulse ??
                throw new ArgumentNullException(
                    nameof(pulse));
        }

        public IPulse Pulse { get; }

        public int WorkUnitCount
        {
            get
            {
                lock (_sync)
                {
                    return _workUnits.Count;
                }
            }
        }

        public IReadOnlyList<IWorkUnit> WorkUnits
        {
            get
            {
                lock (_sync)
                {
                    return _workUnits.ToArray();
                }
            }
        }

        public bool CanResolve
        {
            get
            {
                lock (_sync)
                {
                    return CanResolveCore();
                }
            }
        }

        public bool CanFinalize
        {
            get
            {
                lock (_sync)
                {
                    return CanFinalizeCore();
                }
            }
        }

        public bool AddWorkUnit(
            IWorkUnit workUnit)
        {
            if (workUnit == null)
            {
                throw new ArgumentNullException(
                    nameof(workUnit));
            }

            if (workUnit.PulseId != Pulse.Id)
            {
                return false;
            }

            lock (_sync)
            {
                if (Pulse.IsTerminal)
                {
                    return false;
                }

                if (!_workUnitIds.Add(
                        workUnit.Id))
                {
                    return false;
                }

                _workUnits.Add(
                    workUnit);

                return true;
            }
        }

        public bool TryResolve(
            object? result,
            out IMoment? moment)
        {
            lock (_sync)
            {
                moment = null;

                if (!CanResolveCore())
                {
                    return false;
                }

                return TryCreateCompletedMoment(
                    result,
                    out moment);
            }
        }

        public bool TryFinalize(
            object? result,
            out IMoment? moment)
        {
            lock (_sync)
            {
                moment = null;

                if (!CanFinalizeCore())
                {
                    return false;
                }

                var failedWork =
                    GetFirstFailedWorkUnit();

                if (failedWork != null)
                {
                    return TryCreateFailedMoment(
                        failedWork,
                        out moment);
                }

                if (ContainsCancelledWorkUnit())
                {
                    return TryCreateCancelledMoment(
                        out moment);
                }

                return TryCreateCompletedMoment(
                    result,
                    out moment);
            }
        }

        private bool TryCreateCompletedMoment(
            object? result,
            out IMoment? moment)
        {
            moment = null;

            if (!CanResolveCore())
            {
                return false;
            }

            if (!Pulse.TryComplete())
            {
                return false;
            }

            moment =
                new Moment(
                    Pulse.Id,
                    MomentOutcome.Completed,
                    result,
                    null,
                    GetWorkUnitIds());

            return true;
        }

        private bool TryCreateFailedMoment(
            IWorkUnit failedWork,
            out IMoment? moment)
        {
            moment = null;

            var exception =
                failedWork.Exception;

            if (exception == null)
            {
                return false;
            }

            if (!Pulse.TryFail())
            {
                return false;
            }

            moment =
                new Moment(
                    Pulse.Id,
                    MomentOutcome.Failed,
                    null,
                    exception,
                    GetWorkUnitIds());

            return true;
        }

        private bool TryCreateCancelledMoment(
            out IMoment? moment)
        {
            moment = null;

            if (!Pulse.TryCancel())
            {
                return false;
            }

            moment =
                new Moment(
                    Pulse.Id,
                    MomentOutcome.Cancelled,
                    null,
                    null,
                    GetWorkUnitIds());

            return true;
        }

        private bool CanResolveCore()
        {
            if (Pulse.Status !=
                PulseStatus.Processing)
            {
                return false;
            }

            if (_workUnits.Count == 0)
            {
                return false;
            }

            for (
                var index = 0;
                index < _workUnits.Count;
                index++)
            {
                if (_workUnits[index].Status !=
                    WorkUnitStatus.Completed)
                {
                    return false;
                }
            }

            return true;
        }

        private bool CanFinalizeCore()
        {
            if (Pulse.Status !=
                PulseStatus.Processing)
            {
                return false;
            }

            if (_workUnits.Count == 0)
            {
                return false;
            }

            for (
                var index = 0;
                index < _workUnits.Count;
                index++)
            {
                if (!_workUnits[index].IsTerminal)
                {
                    return false;
                }
            }

            return true;
        }

        private IWorkUnit? GetFirstFailedWorkUnit()
        {
            for (
                var index = 0;
                index < _workUnits.Count;
                index++)
            {
                var workUnit =
                    _workUnits[index];

                if (workUnit.Status ==
                    WorkUnitStatus.Failed)
                {
                    return workUnit;
                }
            }

            return null;
        }

        private bool ContainsCancelledWorkUnit()
        {
            for (
                var index = 0;
                index < _workUnits.Count;
                index++)
            {
                if (_workUnits[index].Status ==
                    WorkUnitStatus.Cancelled)
                {
                    return true;
                }
            }

            return false;
        }

        private Guid[] GetWorkUnitIds()
        {
            return _workUnits
                .Select(
                    workUnit =>
                        workUnit.Id)
                .ToArray();
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using WildsEngine.Core.Abstractions;

namespace WildsEngine.Core.Units
{
    /// <summary>
    /// Immutable record of a terminal Pulse outcome.
    ///
    /// A Moment connects the Pulse identity, contributing
    /// WorkUnits, final result, and terminal outcome.
    /// </summary>
    public sealed class Moment :
        BaseUnit,
        IMoment
    {
        private readonly IReadOnlyList<Guid>
            _workUnitIds;

        public Moment(
            Guid pulseId,
            object? result,
            IEnumerable<Guid>? workUnitIds = null)
            : this(
                Guid.NewGuid(),
                pulseId,
                MomentOutcome.Completed,
                result,
                null,
                workUnitIds)
        {
        }

        public Moment(
            Guid id,
            Guid pulseId,
            object? result,
            IEnumerable<Guid>? workUnitIds = null)
            : this(
                id,
                pulseId,
                MomentOutcome.Completed,
                result,
                null,
                workUnitIds)
        {
        }

        public Moment(
            Guid pulseId,
            MomentOutcome outcome,
            object? result,
            Exception? exception,
            IEnumerable<Guid>? workUnitIds = null)
            : this(
                Guid.NewGuid(),
                pulseId,
                outcome,
                result,
                exception,
                workUnitIds)
        {
        }

        public Moment(
            Guid id,
            Guid pulseId,
            MomentOutcome outcome,
            object? result,
            Exception? exception,
            IEnumerable<Guid>? workUnitIds = null)
            : base(
                id)
        {
            PulseId =
                ValidatePulseId(
                    pulseId);

            ValidateOutcome(
                outcome,

            Result = result;

        public Exception? Exception { get; }

            _workUnitIds;

        private static Guid ValidatePulseId(
            Guid pulseId)
        {
            if (pulseId == Guid.Empty)
            {
                throw new ArgumentException(
                    "Moments require a non-empty Pulse ID.",
                    nameof(pulseId));
            }

            return pulseId;
        }

        private static void ValidateOutcome(
            MomentOutcome outcome,
            Exception? exception)
        {
            if (outcome ==
                MomentOutcome.Failed &&
                exception == null)
            {
                throw new ArgumentException(
                    "Failed Moments require an exception.",
                    nameof(exception));
            }

            if (outcome !=
                MomentOutcome.Failed &&
                exception != null)
            {
                throw new ArgumentException(
                    "Only failed Moments may carry an exception.",
                    nameof(exception));
            }
        }

        private static IReadOnlyList<Guid>
            CopyWorkUnitIds(
                IEnumerable<Guid>? workUnitIds)
        {
            if (workUnitIds == null)
            {
                return Array.Empty<Guid>();
            }

            var copy =
                workUnitIds
                    .Where(
                        id => id != Guid.Empty)
                    .Distinct()
                    .ToArray();

            return copy.Length == 0
                ? Array.Empty<Guid>()
                : copy;
        }

using System;
using QSL.Core;

namespace WildsEngine.Qsl
{
    /// <summary>
    /// Represents an unsuccessful QSL processing result
    /// encountered while resolving a Wilds WorkUnit.
    ///
    /// Throwing this exception allows WildsEngine to record
    /// the QSL failure as a failed WorkUnit, failed Pulse,
    /// and failed Moment.
    /// </summary>
    public sealed class WildsQslExecutionException :
        Exception
    {
        public WildsQslExecutionException(
            QslProcessingResult result)
            : base(
                CreateMessage(
                    result),
                result != null
                    ? result.Exception
                    : null)
        {
            Result = result ??
                throw new ArgumentNullException(
                    nameof(result));
        }

        public QslProcessingResult Result { get; }

        private static string CreateMessage(
            QslProcessingResult result)
        {
            if (result == null)
            {
                return "QSL processing failed without a result.";
            }

            if (result.HasException &&
                result.Exception != null)
            {
                return
                    "QSL processing raised an exception: " +
                    result.Exception.Message;
            }

            if (result.ParseFailed)
            {
                return
                    "QSL parsing failed with " +
                    result.ParserDiagnostics.Count +
                    " diagnostic(s).";
            }

            if (result.EvaluationFailed)
            {
                return
                    "QSL evaluation failed with " +
                    result.RuntimeDiagnostics.Count +
                    " runtime diagnostic(s).";
            }

            return "QSL processing did not succeed.";
        }
    }
}

using System;
using System.Threading;
using System.Threading.Tasks;
using QSL.Core;
using WildsEngine.Core.Abstractions;

namespace WildsEngine.Qsl
{
    /// <summary>
    /// Executes QSL source through the complete WildsEngine
    /// request lifecycle.
    ///
    /// source
    ///     -> Pulse
    ///     -> WorkUnit
    ///     -> QSL evaluation
    ///     -> Moment
    ///     -> Timeline
    ///     -> Epoch
    /// </summary>
    public sealed class WildsQslExecutor
    {
        private readonly QslEngine _qsl;
        private readonly Core.WildsEngine _wilds;

        public WildsQslExecutor(
            QslEngine qsl,
            Core.WildsEngine wilds)
        {
            _qsl = qsl ??
                throw new ArgumentNullException(
                    nameof(qsl));

            _wilds = wilds ??
                throw new ArgumentNullException(
                    nameof(wilds));
        }

        /// <summary>
        /// Executes one complete QSL request through Wilds.
        ///
        /// Wilds remains responsible for the Pulse,
        /// WorkUnit, terminal Moment, Timeline, and Epoch.
        /// QSL remains responsible only for processing the
        /// supplied QSL source.
        /// </summary>
        public ValueTask<IMoment> ExecuteAsync(
            string source,
            CancellationToken cancellationToken = default)
        {
            if (string.IsNullOrWhiteSpace(
                    source))
            {
                throw new ArgumentException(
                    "QSL source cannot be empty.",
                    nameof(source));
            }

            return _wilds.ExecuteAsync(
                source,
                ExecuteQslAsync,
                cancellationToken);
        }

        private ValueTask<object?> ExecuteQslAsync(
            object? request,
            CancellationToken cancellationToken)
        {
            cancellationToken
                .ThrowIfCancellationRequested();

            var source =
                request as string;

            if (source == null)
            {
                throw new InvalidOperationException(
                    "WildsQslExecutor expected a QSL " +
                    "source string.");
            }

            var result =
                _qsl.Evaluate(
                    source);

            if (!result.Succeeded)
            {
                throw new WildsQslExecutionException(
                    result);
            }

            return new ValueTask<object?>(
                result.Value);
        }
    }
}

        private static async Task
            ConcurrentLoadRespectsLimit()
        {
            const int limit = 3;
            const int count = 24;

            using var engine =
                CreateEngine(
                    "certification.concurrent",
                    retainedInstructions: limit,
                    concurrentInstructions: limit);

            Assert(
                engine.Initialize(),
                "Initialize() should succeed.");

            var current = 0;
            var maximum = 0;

            var release =
                new TaskCompletionSource<bool>(
                    TaskCreationOptions
                        .RunContinuationsAsynchronously);

            var entered =
                new TaskCompletionSource<bool>(
                    TaskCreationOptions
                        .RunContinuationsAsynchronously);

            var tasks =
                new List<Task<int>>(
                    count);

            for (var i = 0; i < count; i++)
            {
                tasks.Add(
                    engine.ExecuteInstructionAsync<int>(
                            async (_, token) =>
                            {
                                var now =
                                    Interlocked.Increment(
                                        ref current);

                                UpdateMaximum(
                                    ref maximum,
                                    now);

                                if (now >= limit)
                                {
                                    entered.TrySetResult(
                                        true);
                                }

                                try
                                {
                                    await release.Task
                                        .ConfigureAwait(false);

                                    token.ThrowIfCancellationRequested();

                                    return 1;
                                }
                                finally
                                {
                                    Interlocked.Decrement(
                                        ref current);
                                }
                            })
                        .AsTask());
            }

            await entered.Task
                .ConfigureAwait(false);

            LogPool(
                engine,
                "At concurrency limit");

            Assert(
                engine.Instructions.ActiveCount <= limit,
                "Active leases exceeded MaximumConcurrent.");

            release.TrySetResult(
                true);

            await Task.WhenAll(
                    tasks)
                .ConfigureAwait(false);

            Assert(
                maximum == limit,
                "Concurrent load did not exercise the configured limit.");

            Assert(
                engine.Instructions.ActiveCount == 0,
                "All concurrent leases should be returned.");

            Assert(
                engine.Instructions.CreatedCount <= limit,
                "Pool created more processors than " +
                "concurrent leasing permits.");
        }

        private static void UpdateMaximum(
            ref int target,
            int value)
        {
            while (true)
            {
                var current =
                    Volatile.Read(
                        ref target);

                if (value <= current)
                    return;

                if (Interlocked.CompareExchange(
                        ref target,
                        value,
                        current) == current)
                {
                    return;
                }
            }
        }

using System.Text.Json;
using QSL.Core;
using QSL.Core.Provenance;

internal static class CertificationRunner
{
    private const string SuiteFileName =
        "qsl_statement_testing.json";

    private const string ReportFileName =
        "qsl_statement_testing_output.json";

    public static int Run()
    {
        var suitePath =
            Path.Combine(
                AppContext.BaseDirectory,
                "Certification",
                SuiteFileName
            );

        if (!File.Exists(suitePath))
        {
            Console.Error.WriteLine(
                "Certification suite was not found:"
            );

            Console.Error.WriteLine(
                suitePath
            );

            return 2;
        }

        QslCertificationSuite? suite;

        try
        {
            var json =
                File.ReadAllText(
                    suitePath
                );

            suite =
                JsonSerializer.Deserialize
                    <QslCertificationSuite>(
                        json,
                        new JsonSerializerOptions
                        {
                            PropertyNameCaseInsensitive =
                                true
                        }
                    );
        }
        catch (Exception exception)
        {
            Console.Error.WriteLine(
                "Certification suite could not be loaded."
            );

            Console.Error.WriteLine(
                exception.Message
            );

            return 2;
        }

        if (suite is null ||
            suite.Tests.Count == 0)
        {
            Console.Error.WriteLine(
                "Certification suite contains no tests."
            );

            return 2;
        }

        var host =
            QslCliHost.Create();

        var report =
            new QslCertificationReport();

        Console.WriteLine(
            "========================================"
        );

        Console.WriteLine(
            "QSL CERTIFICATION"
        );

        Console.WriteLine(
            "========================================"
        );

        Console.WriteLine(
            "Suite: " +
            suitePath
        );

        Console.WriteLine(
            "Tests: " +
            suite.Tests.Count
        );

        Console.WriteLine();

        for (
            var index = 0;
            index < suite.Tests.Count;
            index++)
        {
            var test =
                suite.Tests[index];

            var result =
                ExecuteTest(
                    host.Engine,
                    test
                );

            report.Results.Add(
                result
            );

            if (result.Passed)
            {
                report.Passed++;
            }
            else
            {
                report.Failed++;
            }

            Console.Write(
                result.Passed
                    ? "."
                    : "F"
            );

            if ((index + 1) % 50 == 0)
            {
                Console.WriteLine();
            }
        }

        if (suite.Tests.Count % 50 != 0)
        {
            Console.WriteLine();
        }

        report.Total =
            report.Results.Count;

        Console.WriteLine();
        Console.WriteLine(
            "========================================"
        );

        Console.WriteLine(
            "Total:  " +
            report.Total
        );

        Console.WriteLine(
            "Passed: " +
            report.Passed
        );

        Console.WriteLine(
            "Failed: " +
            report.Failed
        );

        Console.WriteLine(
            "========================================"
        );

        if (report.Failed > 0)
        {
            WriteFailures(
                report.Results
            );
        }

        var reportPath =
            Path.Combine(
                Directory.GetCurrentDirectory(),
                ReportFileName
            );

        try
        {
            var reportJson =
                JsonSerializer.Serialize(
                    report,
                    new JsonSerializerOptions
                    {
                        WriteIndented =
                            true
                    }
                );

            File.WriteAllText(
                reportPath,
                reportJson
            );

            Console.WriteLine();
            Console.WriteLine(
                "Report:"
            );

            Console.WriteLine(
                reportPath
            );
        }
        catch (Exception exception)
        {
            Console.Error.WriteLine();
            Console.Error.WriteLine(
                "Certification report could not be written."
            );

            Console.Error.WriteLine(
                exception.Message
            );

            return 3;
        }

        return report.Failed == 0
            ? 0
            : 1;
    }

    private static QslCertificationResult ExecuteTest(
        QslEngine engine,
        QslCertificationTest test)
    {
        var processingResult =
            engine.Evaluate(
                test.Statement
            );

        var output =
            QslValueFormatter.Format(
                processingResult.Value
            );

        var expectsError =
            string.Equals(
                test.Category,
                "Error",
                StringComparison.OrdinalIgnoreCase
            );

        var passed =
            expectsError
                ? !processingResult.Succeeded
                : processingResult.Succeeded &&
                  string.Equals(
                      output,
                      test.Expected,
                      StringComparison.Ordinal
                  );

        var runtimeDiagnostics =
            new string[
                processingResult
                    .RuntimeDiagnostics
                    .Count
            ];

        for (
            var index = 0;
            index < runtimeDiagnostics.Length;
            index++)
        {
            runtimeDiagnostics[index] =
                FormatRuntimeDiagnostic(
                    processingResult
                        .RuntimeDiagnostics[index]
                );
        }

        var parserDiagnostics =
            new string[
                processingResult
                    .ParserDiagnostics
                    .Count
            ];

        for (
            var index = 0;
            index < parserDiagnostics.Length;
            index++)
        {
            parserDiagnostics[index] =
                processingResult
                    .ParserDiagnostics[index];
        }

        return new QslCertificationResult
        {
            Category =
                test.Category,

            Statement =
                test.Statement,

            Expected =
                test.Expected,

            Output =
                output,

            Passed =
                passed,

            Succeeded =
                processingResult.Succeeded,

            ParseFailed =
                processingResult.ParseFailed,

            EvaluationFailed =
                processingResult.EvaluationFailed,

            ParserDiagnostics =
                parserDiagnostics,

            RuntimeDiagnostics =
                runtimeDiagnostics,

            Exception =
                processingResult
                    .Exception?
                    .Message
        };
    }

    private static string FormatRuntimeDiagnostic(
        QslProvenanceEntry entry)
    {
        return
            "[" +
            entry.EntryType +
            "] " +
            entry.Source +
            ": " +
            entry.Message;
    }

    private static void WriteFailures(
        IReadOnlyList<QslCertificationResult>
            results)
    {
        Console.WriteLine();
        Console.WriteLine(
            "Failures:"
        );

        foreach (var result in results)
        {
            if (result.Passed)
            {
                continue;
            }

            Console.WriteLine();
            Console.WriteLine(
                "[" +
                result.Category +
                "] " +
                result.Statement
            );

            Console.WriteLine(
                "  Expected: " +
                result.Expected
            );

            Console.WriteLine(
                "  Actual:   " +
                (
                    string.IsNullOrEmpty(
                        result.Output
                    )
                        ? "<null>"
                        : result.Output
                )
            );

            Console.WriteLine(
                "  Succeeded: " +
                result.Succeeded
            );

            foreach (var diagnostic in
                     result.ParserDiagnostics)
            {
                Console.WriteLine(
                    "  Parser: " +
                    diagnostic
                );
            }

            foreach (var diagnostic in
                     result.RuntimeDiagnostics)
            {
                Console.WriteLine(
                    "  Runtime: " +
                    diagnostic
                );
            }

            if (!string.IsNullOrEmpty(
                    result.Exception))
            {
                Console.WriteLine(
                    "  Exception: " +
                    result.Exception
                );
            }
        }
    }
}
