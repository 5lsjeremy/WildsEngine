    1: using System;
    2: using WildsEngine.Core.Abstractions;
    3: using WildsEngine.Core.Abstractions.Neighbors.Propagation;
    4: 
    5: namespace WildsEngine.Core.Neighbors.Propagation
    6: {
    7:     public sealed class ExponentialDistancePropagation :
    8:         INeighborPropagation
    9:     {
   10:         private readonly double _decay;
   11: 
   12:         public ExponentialDistancePropagation(
   13:             double decay)
   14:         {
   15:             ValidateNonNegativeFinite(
   16:                 decay,
   17:                 nameof(decay));
   18: 
   19:             _decay =
   20:                 decay;
   21:         }
   22: 
   23:         public double Propagate(
   24:             double intensity,
   25:             INeighborRelationship relationship)
   26:         {
   27:             if (relationship == null)
   28:             {
   29:                 throw new ArgumentNullException(
   30:                     nameof(relationship));
   31:             }
   32: 
   33:             ValidateNonNegativeFinite(
   34:                 intensity,
   35:                 nameof(intensity));
   36: 
   37:             var distance =
   38:                 relationship.Distance;
   39: 
   40:             ValidateNonNegativeFinite(
   41:                 distance,
   42:                 nameof(relationship));
   43: 
   44:             var exponent =
   45:                 -_decay *
   46:                 distance;
   47: 
   48:             /*
   49:              * Very large finite decay and distance values can
   50:              * overflow their product to negative infinity.
   51:              *
   52:              * For non-negative decay and distance, that limit
   53:              * legitimately represents complete attenuation:
   54:              *
   55:              *     Exp(-Infinity) = 0
   56:              *
   57:              * NaN remains invalid and must never pass through.
   58:              */
   59:             if (double.IsNaN(
   60:                     exponent))
   61:             {
   62:                 throw new ArithmeticException(
   63:                     "Propagation produced an invalid exponent.");
   64:             }
   65: 
   66:             var attenuation =
   67:                 Math.Exp(
   68:                     exponent);
   69: 
   70:             if (double.IsNaN(
   71:                     attenuation) ||
   72:                 double.IsInfinity(
   73:                     attenuation))
   74:             {
   75:                 throw new ArithmeticException(
   76:                     "Propagation produced non-finite attenuation.");
   77:             }
   78: 
   79:             var propagated =
   80:                 intensity *
   81:                 attenuation;
   82: 
   83:             if (double.IsNaN(
   84:                     propagated) ||
   85:                 double.IsInfinity(
   86:                     propagated))
   87:             {
   88:                 throw new ArithmeticException(
   89:                     "Propagation produced a non-finite result.");
   90:             }
   91: 
   92:             return propagated;
   93:         }
   94: 
   95:         private static void ValidateNonNegativeFinite(
   96:             double value,
   97:             string parameterName)
   98:         {
   99:             if (double.IsNaN(
  100:                     value) ||
  101:                 double.IsInfinity(
  102:                     value) ||
  103:                 value < 0.0)
  104:             {
  105:                 throw new ArgumentOutOfRangeException(
  106:                     parameterName);
  107:             }
  108:         }
  109:     }
  110: }
