using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Linq;
using WildsEngine.Core.Abstractions;
using WildsEngine.Core.Objects;

namespace WildsEngine.Core.Processing
{
    public abstract class RegistryBase<T> :
        BaseObject,
        IRegistry<T>
        where T : class
    {
        private readonly ConcurrentDictionary<Guid, T> _items =
            new ConcurrentDictionary<Guid, T>();

        private readonly IComponent _component;
        private readonly Func<T, Guid> _identity;

        protected RegistryBase(
            IComponent component,
            string layer,
            Func<T, Guid> identity)
            : base(layer)
        {
            _component = component ??
                throw new ArgumentNullException(
                    nameof(component));

            _identity = identity ??
                throw new ArgumentNullException(
                    nameof(identity));
        }

        public sealed override IComponent Component =>
            _component;

        public int Count =>
            _items.Count;

        public IReadOnlyCollection<T> Items =>
            _items.Values.ToArray();

        public bool Register(T item)
        {
            if (item == null)
            {
                throw new ArgumentNullException(
                    nameof(item));
            }

            var id =
                _identity(item);

            if (id == Guid.Empty)
                return false;

            return _items.TryAdd(
                id,
                item);
        }

        public bool Contains(Guid id)
        {
            if (id == Guid.Empty)
                return false;

            return _items.ContainsKey(
                id);
        }

        public T? Get(Guid id)
        {
            if (id == Guid.Empty)
                return null;

            return _items.TryGetValue(
                id,
                out var item)
                    ? item
                    : null;
        }

        public bool TryGet(
            Guid id,
            out T? item)
        {
            if (id == Guid.Empty)
            {
                item = null;

                return false;
            }

            return _items.TryGetValue(
                id,
                out item);
        }

        public bool Remove(Guid id)
        {
            if (id == Guid.Empty)
                return false;

            return _items.TryRemove(
                id,
                out _);
        }

        public void Clear()
        {
            _items.Clear();
        }

        public bool IsInitialized { get; private set; }

        public bool Initialize()
        {
            if (IsInitialized)
                return false;

            OnInitialize();

            IsInitialized = true;

            return true;
        }

        public bool Shutdown()
        {
            if (!IsInitialized)
                return false;

            OnShutdown();

            IsInitialized = false;

            return true;
        }

        protected abstract void OnInitialize();

        protected abstract void OnShutdown();
    }
}

using System;
using System.Collections.Generic;
using WildsEngine.Core.Abstractions;

namespace WildsEngine.Core.Processing
{
    public abstract class StateInstructionBase :
        IStateInstruction
    {
        private readonly Dictionary<string, object?> _arguments;
        private readonly Dictionary<string, object?> _metadata;

        protected StateInstructionBase(
            string intent,
            object? payload = null,
            IEnumerable<KeyValuePair<string, object?>>? arguments = null,
            IEnumerable<KeyValuePair<string, object?>>? metadata = null)
        {
            if (string.IsNullOrWhiteSpace(intent))
            {
                throw new ArgumentException(
                    "Instruction intent cannot be empty.",
                    nameof(intent));
            }

            Id = Guid.NewGuid();
            Intent = intent;
            Payload = payload;

            _arguments = CreateDictionary(arguments);
            _metadata = CreateDictionary(metadata);
        }

        public Guid Id { get; }

        public string Intent { get; }

        public object? Payload { get; }

        public IReadOnlyDictionary<string, object?> Arguments =>
            _arguments;

        public IReadOnlyDictionary<string, object?> Metadata =>
            _metadata;

        protected void SetArgument(
            string key,
            object? value)
        {
            ValidateKey(key, nameof(key));

            _arguments[key] = value;
        }

        protected void SetMetadata(
            string key,
            object? value)
        {
            ValidateKey(key, nameof(key));

            _metadata[key] = value;
        }

        protected bool RemoveArgument(string key)
        {
            ValidateKey(key, nameof(key));

            return _arguments.Remove(key);
        }

        protected bool RemoveMetadata(string key)
        {
            ValidateKey(key, nameof(key));

            return _metadata.Remove(key);
        }

        private static Dictionary<string, object?> CreateDictionary(
            IEnumerable<KeyValuePair<string, object?>>? source)
        {
            var result =
                new Dictionary<string, object?>();

            if (source == null)
            {
                return result;
            }

            foreach (var pair in source)
            {
                ValidateKey(pair.Key, nameof(source));

                result[pair.Key] = pair.Value;
            }

            return result;
        }

        private static void ValidateKey(
            string key,
            string parameterName)
        {
            if (string.IsNullOrWhiteSpace(key))
            {
                throw new ArgumentException(
                    "Instruction keys cannot be empty.",
                    parameterName);
            }
        }
    }
}

using System;
using System.Diagnostics;
using WildsEngine.Core.Abstractions;
using WildsEngine.Core.Pooling;
using WildsEngine.Core.Processing.Processor;

namespace WildsEngine.Benchmarks
{
    /// <summary>
    /// Compares direct TaskProcessor allocation against
    /// pooled TaskProcessor leasing.
    /// </summary>
    public sealed class ProcessorPoolingBenchmark
    {
        private readonly IComponent _component;
        private readonly ITaskRegistry _registry;

        public ProcessorPoolingBenchmark(
            IComponent component,
            ITaskRegistry registry)
        {
            _component = component ??
                throw new ArgumentNullException(
                    nameof(component));

            _registry = registry ??
                throw new ArgumentNullException(
                    nameof(registry));
        }

        public ProcessorPoolingBenchmarkResult Run(
            int iterations,
            int maximumRetained = 64,
            int maximumConcurrent = 64)
        {
            if (iterations <= 0)
            {
                throw new ArgumentOutOfRangeException(
                    nameof(iterations));
            }

            if (maximumRetained < 0)
            {
                throw new ArgumentOutOfRangeException(
                    nameof(maximumRetained));
            }

            if (maximumConcurrent <= 0)
            {
                throw new ArgumentOutOfRangeException(
                    nameof(maximumConcurrent));
            }

            WarmUp();

            var allocationMeasurementAvailable =
                IsAllocationMeasurementAvailable();

            var directAllocatedBytesBefore =
                GetAllocatedBytes();

            var directWatch =
                Stopwatch.StartNew();

            for (var i = 0; i < iterations; i++)
            {
                var processor =
                    new TaskProcessor(
                        _component,
                        "benchmark.direct",
                        _registry);

                Consume(
                    processor);
            }

            directWatch.Stop();

            var directAllocatedBytes =
                allocationMeasurementAvailable
                    ? GetAllocatedBytes() -
                      directAllocatedBytesBefore
                    : 0;

            using (var pool =
                new ObjectPool<ITaskProcessor>(
                    factory: () =>
                        new TaskProcessor(
                            _component,
                            "benchmark.pooled",
                            _registry),
                    maximumRetained:
                        maximumRetained,
                    maximumConcurrent:
                        maximumConcurrent))
            {
                var pooledAllocatedBytesBefore =
                    GetAllocatedBytes();

                var pooledWatch =
                    Stopwatch.StartNew();

                for (var i = 0; i < iterations; i++)
                {
                    using (var lease = pool.Rent())
                    {
                        Consume(
                            lease.Instance);
                    }
                }

                pooledWatch.Stop();

                var pooledAllocatedBytes =
                    allocationMeasurementAvailable
                        ? GetAllocatedBytes() -
                          pooledAllocatedBytesBefore
                        : 0;

                return new ProcessorPoolingBenchmarkResult(
                    iterations,
                    directWatch.Elapsed,
                    pooledWatch.Elapsed,
                    directAllocatedBytes,
                    pooledAllocatedBytes,
                    pool.CreatedCount,
                    allocationMeasurementAvailable);
            }
        }

        private void WarmUp()
        {
            for (var i = 0; i < 128; i++)
            {
                var processor =
                    new TaskProcessor(
                        _component,
                        "benchmark.warmup",
                        _registry);

                Consume(
                    processor);
            }
        }

        private static bool IsAllocationMeasurementAvailable()
        {
#if NETSTANDARD2_1
            return true;
#else
            return false;
#endif
        }

        private static long GetAllocatedBytes()
        {
#if NETSTANDARD2_1
            return GC.GetAllocatedBytesForCurrentThread();
#else
            return 0;
#endif
        }

        private static void Consume(
            ITaskProcessor processor)
        {
            if (processor == null)
            {
                throw new InvalidOperationException();
            }
        }
    }
}

using System;

namespace WildsEngine.Benchmarks
{
    /// <summary>
    /// Contains the results of a processor pooling benchmark.
    /// </summary>
    public sealed class ProcessorPoolingBenchmarkResult
    {
        public ProcessorPoolingBenchmarkResult(
            int iterations,
            TimeSpan directDuration,
            TimeSpan pooledDuration,
            long directAllocatedBytes,
            long pooledAllocatedBytes,
            int pooledInstancesCreated,
            bool allocationMeasurementAvailable)
        {
            Iterations = iterations;
            DirectDuration = directDuration;
            PooledDuration = pooledDuration;
            DirectAllocatedBytes =
                directAllocatedBytes;
            PooledAllocatedBytes =
                pooledAllocatedBytes;
            PooledInstancesCreated =
                pooledInstancesCreated;
            AllocationMeasurementAvailable =
                allocationMeasurementAvailable;
        }

        public int Iterations { get; }

        public TimeSpan DirectDuration { get; }

        public TimeSpan PooledDuration { get; }

        public long DirectAllocatedBytes { get; }

        public long PooledAllocatedBytes { get; }

        public int PooledInstancesCreated { get; }

        public bool AllocationMeasurementAvailable { get; }

        public override string ToString()
        {
            var allocations =
                AllocationMeasurementAvailable
                    ? $"Direct allocation: {DirectAllocatedBytes} bytes{Environment.NewLine}" +
                      $"Pooled allocation: {PooledAllocatedBytes} bytes{Environment.NewLine}"
                    : $"Allocation measurement: unavailable{Environment.NewLine}";

            return
                $"Iterations: {Iterations}{Environment.NewLine}" +
                $"Direct duration: {DirectDuration.TotalMilliseconds:F3} ms{Environment.NewLine}" +
                $"Pooled duration: {PooledDuration.TotalMilliseconds:F3} ms{Environment.NewLine}" +
                allocations +
                $"Pooled processors created: {PooledInstancesCreated}";
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using WildsEngine.Core.Abstractions;
using WildsEngine.Core.Units;

namespace WildsEngine.Core.Processing.Coordinator
{
    /// <summary>
    /// Coordinates all required WorkUnits for one Pulse.
    ///
    /// The coordinator prevents early completion and records
    /// successful, failed, and cancelled Pulse outcomes.
    /// </summary>
    public sealed class PulseCoordinator :
        IPulseCoordinator
    {
        private readonly object _sync =
            new object();

        private readonly List<IWorkUnit> _workUnits =
            new List<IWorkUnit>();

        private readonly HashSet<Guid> _workUnitIds =
            new HashSet<Guid>();

        public PulseCoordinator(
            IPulse pulse)
        {
            Pulse = pulse ??
                throw new ArgumentNullException(
                    nameof(pulse));
        }

        public IPulse Pulse { get; }

        public int WorkUnitCount
        {
            get
            {
                lock (_sync)
                {
                    return _workUnits.Count;
                }
            }
        }

        public IReadOnlyList<IWorkUnit> WorkUnits
        {
            get
            {
                lock (_sync)
                {
                    return _workUnits.ToArray();
                }
            }
        }

        public bool CanResolve
        {
            get
            {
                lock (_sync)
                {
                    return CanResolveCore();
                }
            }
        }

        public bool CanFinalize
        {
            get
            {
                lock (_sync)
                {
                    return CanFinalizeCore();
                }
            }
        }

        public bool AddWorkUnit(
            IWorkUnit workUnit)
        {
            if (workUnit == null)
            {
                throw new ArgumentNullException(
                    nameof(workUnit));
            }

            if (workUnit.PulseId != Pulse.Id)
            {
                return false;
            }

            lock (_sync)
            {
                if (Pulse.IsTerminal)
                {
                    return false;
                }

                if (!_workUnitIds.Add(
                        workUnit.Id))
                {
                    return false;
                }

                _workUnits.Add(
                    workUnit);

                return true;
            }
        }

        public bool TryResolve(
            object? result,
            out IMoment? moment)
        {
            lock (_sync)
            {
                moment = null;

                if (!CanResolveCore())
                {
                    return false;
                }

                return TryCreateCompletedMoment(
                    result,
                    out moment);
            }
        }

        public bool TryFinalize(
            object? result,
            out IMoment? moment)
        {
            lock (_sync)
            {
                moment = null;

                if (!CanFinalizeCore())
                {
                    return false;
                }

                var failedWork =
                    GetFirstFailedWorkUnit();

                if (failedWork != null)
                {
                    return TryCreateFailedMoment(
                        failedWork,
                        out moment);
                }

                if (ContainsCancelledWorkUnit())
                {
                    return TryCreateCancelledMoment(
                        out moment);
                }

                return TryCreateCompletedMoment(
                    result,
                    out moment);
            }
        }

        private bool TryCreateCompletedMoment(
            object? result,
            out IMoment? moment)
        {
            moment = null;

            if (!CanResolveCore())
            {
                return false;
            }

            if (!Pulse.TryComplete())
            {
                return false;
            }

            moment =
                new Moment(
                    Pulse.Id,
                    MomentOutcome.Completed,
                    result,
                    null,
                    GetWorkUnitIds());

            return true;
        }

        private bool TryCreateFailedMoment(
            IWorkUnit failedWork,
            out IMoment? moment)
        {
            moment = null;

            var exception =
                failedWork.Exception;

            if (exception == null)
            {
                return false;
            }

            if (!Pulse.TryFail())
            {
                return false;
            }

            moment =
                new Moment(
                    Pulse.Id,
                    MomentOutcome.Failed,
                    null,
                    exception,
                    GetWorkUnitIds());

            return true;
        }

        private bool TryCreateCancelledMoment(
            out IMoment? moment)
        {
            moment = null;

            if (!Pulse.TryCancel())
            {
                return false;
            }

            moment =
                new Moment(
                    Pulse.Id,
                    MomentOutcome.Cancelled,
                    null,
                    null,
                    GetWorkUnitIds());

            return true;
        }

        private bool CanResolveCore()
        {
            if (Pulse.Status !=
                PulseStatus.Processing)
            {
                return false;
            }

            if (_workUnits.Count == 0)
            {
                return false;
            }

            for (
                var index = 0;
                index < _workUnits.Count;
                index++)
            {
                if (_workUnits[index].Status !=
                    WorkUnitStatus.Completed)
                {
                    return false;
                }
            }

            return true;
        }

        private bool CanFinalizeCore()
        {
            if (Pulse.Status !=
                PulseStatus.Processing)
            {
                return false;
            }

            if (_workUnits.Count == 0)
            {
                return false;
            }

            for (
                var index = 0;
                index < _workUnits.Count;
                index++)
            {
                if (!_workUnits[index].IsTerminal)
                {
                    return false;
                }
            }

            return true;
        }

        private IWorkUnit? GetFirstFailedWorkUnit()
        {
            for (
                var index = 0;
                index < _workUnits.Count;
                index++)
            {
                var workUnit =
                    _workUnits[index];

                if (workUnit.Status ==
                    WorkUnitStatus.Failed)
                {
                    return workUnit;
                }
            }

            return null;
        }

        private bool ContainsCancelledWorkUnit()
        {
            for (
                var index = 0;
                index < _workUnits.Count;
                index++)
            {
                if (_workUnits[index].Status ==
                    WorkUnitStatus.Cancelled)
                {
                    return true;
                }
            }

            return false;
        }

        private Guid[] GetWorkUnitIds()
        {
            return _workUnits
                .Select(
                    workUnit =>
                        workUnit.Id)
                .ToArray();
        }
    }
}

    {

        public AddNeighborInstruction(

                    distance),

        {

                ValidateMembership(

        public IBaseObject Neighbor { get; }

        private static IBaseObject ValidateNeighbor(

        public InstructionsProcessor(
            IComponent component,

            IInstructionsRegistry registry,
            IObjectPool<ITaskProcessor> taskProcessors)
            : base(
                component,
                layer)
        {
            Registry = registry ??
                       throw new ArgumentNullException(
                           nameof(registry));

            _taskProcessors = taskProcessors ??
                              throw new ArgumentNullException(
                                  nameof(taskProcessors));
        }

        public IInstructionsRegistry Registry { get; }

        /// <summary>
        /// Rents a task processor for delegated task processing.
        /// The caller owns the returned lease and must dispose it.
        /// </summary>

        {
            return _taskProcessors.RentAsync(
                cancellationToken);

        /// Clears transient processor state before this processor
        /// is returned to its pool.

        {
            return true;
        }
    }

using System;
using WildsEngine.Core.Abstractions;

namespace WildsEngine.Core.Processing.Processor
{
    /// <summary>
    /// Processes task work using a task registry.
    /// Provides the task lifecycle boundary used by
    /// pooled task processors.
    /// </summary>
    public class TaskProcessor :
        ProcessorBase,
        ITaskProcessor
    {
        public TaskProcessor(
            IComponent component,
            string layer,
            ITaskRegistry registry)
            : base(
                component,
                layer)
        {
            Registry = registry ??
                throw new ArgumentNullException(
                    nameof(registry));
        }

        public ITaskRegistry Registry { get; }

        /// <summary>
        /// Parks a task for processing.
        /// </summary>
        public bool RegisterTask(
            ITask task)
        {
            if (task == null)
            {
                throw new ArgumentNullException(
                    nameof(task));
            }

            return Registry.Register(
                task);
        }

        /// <summary>
        /// Determines whether a task is currently parked.
        /// </summary>
        public bool ContainsTask(
            Guid id)
        {
            return Registry.Contains(
                id);
        }

        /// <summary>
        /// Retrieves a parked task by identity.
        /// </summary>
        public ITask? GetTask(
            Guid id)
        {
            return Registry.Get(
                id);
        }

        /// <summary>
        /// Attempts to retrieve a parked task by identity.
        /// </summary>
        public bool TryGetTask(
            Guid id,
            out ITask? task)
        {
            return Registry.TryGet(
                id,
                out task);
        }

        /// <summary>
        /// Removes a parked task after processing completes.
        /// </summary>
        public bool RemoveTask(
            Guid id)
        {
            return Registry.Remove(
                id);
        }

        /// <summary>
        /// Clears transient processor state before this
        /// processor is returned to its pool.
        ///
        /// Registry contents intentionally survive processor
        /// reuse because task lifetime is independent from
        /// processor lifetime.
        /// </summary>
        protected override bool OnReset()
        {
            return true;
        }
    }
}
