                "alpha",
                2.0);

                beta,
                "beta",
                1.0);

            Assert.True(
                emitter.TryEmit(
                    "signal",
                    out var baseEmission));

            var topologyScalar =
                emitter.ResolveTestTopologyScalar(

        [Fact]
        public void Complete_emission_route_resolves_in_one_operation()
        {
            InitializeEmissionReport();

            TestLog.Section(
                "EMITTER GATE 3 - COMPLETE EMISSION RESOLUTION");

            TestLog.Test(
                "Consumer value and current topology resolve into one emission.");

            var component =
                new TestComponent(
                    "gate3.component");

            var emitter =
                WildsFixtureFactory.CreateEmitter(
                    component,
                    properties:
                        new Dictionary<string, object?>
                        {
                            ["signal"] = 1.6
                        });

            var alphaA =
                WildsFixtureFactory.CreateTopologyObject(
                    component,
                    "gate3.alpha.a");

            var alphaB =
                WildsFixtureFactory.CreateTopologyObject(
                    component,
                    "gate3.alpha.b");

            var beta =
                WildsFixtureFactory.CreateTopologyObject(
                    component,
                    "gate3.beta");

            TestLog.Task(
                "Building emitter topology.");

            Assert.True(
                emitter.Neighbors.Add(
                    alphaA,
                    "alpha",
                    1.0));

            Assert.True(
                emitter.Neighbors.Add(
                    alphaB,
                    "alpha",
                    2.0));

            Assert.True(
                emitter.Neighbors.Add(
                    beta,
                    "beta",
                    1.0));

            TestLog.State(
                $"Neighbor relationships = {emitter.Neighbors.Count}");

            TestLog.Task(
                "Resolving base emission through alpha topology.");

            var alphaRelationships =
                emitter.ApplyNeighborOperator(
                    new MembershipNeighborFilter(
                        "alpha"));

            var topologyScalar =
                new NeighborCount().Apply(
                    emitter,
                    alphaRelationships);

            var resolved =
                emitter.TryResolveTestEmission(
                    "signal",
                    new MembershipNeighborFilter(
                        "alpha"),
                    new NeighborCount(),
                    count => count,
                    new TestAdditiveEmissionComposition(),
                    out var value);

            TestLog.State(
                "Base emission = 1.6");

            TestLog.State(
                "Selected membership = alpha");

            TestLog.State(
                $"Topology scalar = {topologyScalar}");

            TestLog.State(
                $"Resolved emission = {value}");

            Assert.True(
                resolved);

            Assert.Equal(
                2,
                topologyScalar);

            Assert.Equal(
                3.6,
                value,
                10);

            TestLog.Pass(
                "Complete emission route resolved successfully.");

            PublishLatestReport();

            WriteFinished(
                "COMPLETE EMISSION RESOLUTION");
        }

using System;
using WildsEngine.Core.Abstractions;

namespace WildsEngine.Core.Emission
{
    /// <summary>
    /// Base implementation for emission composition policies.
    ///
    /// Supplies finite-value validation around the composition
    /// operation while leaving the mathematical relationship
    /// between emission and topology to derived implementations.
    /// </summary>
    public abstract class EmissionCompositionBase :
        IEmissionComposition
    {
        public double Compose(
            double emission,
            double topology)
        {
            ValidateFinite(
                emission,
                nameof(emission));

            ValidateFinite(
                topology,
                nameof(topology));

            var resolved =
                ComposeCore(
                    emission,
                    topology);

            if (double.IsNaN(resolved) ||
                double.IsInfinity(resolved))
            {
                throw new ArithmeticException(
                    "Emission composition must produce a finite scalar.");
            }

            return resolved;
        }

        protected abstract double ComposeCore(
            double emission,
            double topology);

        private static void ValidateFinite(
            double value,
            string parameterName)
        {
            if (double.IsNaN(value) ||
                double.IsInfinity(value))
            {
                throw new ArgumentOutOfRangeException(
                    parameterName,
                    value,
                    "Emission composition values must be finite.");
            }
        }
    }
}

    /// Resolves ordinary finite numeric consumer values into

    ///
    /// No semantic conversion is performed for non-numeric
    /// consumer values.
    /// </summary>
    public sealed class EmissionScalar :
        IEmissionScalar
    {
        public double Resolve(
            object? value)
        {
            if (value == null)
            {
                throw new ArgumentNullException(
                    nameof(value));
            }

            double scalar;

            switch (value)
            {
                case byte number:
                    scalar = number;
                    break;

                case sbyte number:
                    scalar = number;
                    break;

                case short number:
                    scalar = number;
                    break;

                case ushort number:
                    scalar = number;
                    break;

                case int number:
                    scalar = number;
                    break;

                case uint number:
                    scalar = number;
                    break;

                case long number:
                    scalar = number;
                    break;

                case ulong number:
                    scalar = number;
                    break;

                case float number:
                    scalar = number;
                    break;

                case double number:
                    scalar = number;
                    break;

                case decimal number:
                    scalar =
                        Convert.ToDouble(
                            number,
                            CultureInfo.InvariantCulture);
                    break;

                default:
                    throw new ArgumentException(
                        "Emission scalar values must be numeric.",
                        nameof(value));
            }

            if (double.IsNaN(scalar) ||
                double.IsInfinity(scalar))
            {
                throw new ArgumentOutOfRangeException(
                    nameof(value),
                    scalar,
                    "Emission scalar values must be finite.");
            }

            return scalar;
        }
    }
}

using System;
using System.Collections.Generic;
using WildsEngine.Core.Abstractions;
using WildsEngine.Core.Objects;

namespace WildsEngine.Core.Emission
{
    /// <summary>
    /// Base implementation for identifiable Wilds objects
    /// capable of resolving consumer-owned data into scalar
    /// emissions.
    ///
    /// The emitter owns no consumer vocabulary. Property
    /// names, group names, and their values remain defined
    /// by the consumer.
    ///
    /// As a BaseObject, every emitter also participates in
    /// ordinary Wilds neighbor topology.
    /// </summary>
    public abstract class EmitterBase :
        BaseObject,
        IEmitter
    {
        private readonly IComponent _component;
        private readonly IEmissionScalar _scalar;

        protected EmitterBase(
            IComponent component,
            string layer,
            IEmissionDataSource dataSource,
            IEmissionScalar scalar)
            : base(layer)
        {
            _component =
                component ??
                throw new ArgumentNullException(
                    nameof(component));

            DataSource =
                dataSource ??
                throw new ArgumentNullException(
                    nameof(dataSource));

            _scalar =
                scalar ??
                throw new ArgumentNullException(
                    nameof(scalar));
        }

        public override IComponent Component =>
            _component;

        public IEmissionDataSource DataSource { get; }

        protected IEmissionScalar Scalar =>
            _scalar;

        public virtual bool TryEmit(
            string propertyName,
            out double value)
        {
            value =
                0.0;

            if (string.IsNullOrWhiteSpace(
                    propertyName))
            {
                return false;
            }

            if (!DataSource.TryGetProperty(
                    propertyName,
                    out var rawValue))
            {
                return false;
            }

            value =
                ResolveScalar(
                    rawValue);

            return true;
        }

        public virtual bool TryEmitGroup(
            string groupName,
            out IReadOnlyDictionary<string, double> values)
        {
            values =
                new Dictionary<string, double>();

            if (string.IsNullOrWhiteSpace(
                    groupName))
            {
                return false;
            }

            if (!DataSource.TryGetGroup(
                    groupName,
                    out var rawValues))
            {
                return false;
            }

            var resolved =
                new Dictionary<string, double>(
                    StringComparer.Ordinal);

            foreach (var pair in rawValues)
            {
                if (string.IsNullOrWhiteSpace(
                        pair.Key))
                {
                    throw new InvalidOperationException(
                        "Emission groups cannot contain " +
                        "an empty property name.");
                }

                resolved.Add(
                    pair.Key,
                    ResolveScalar(
                        pair.Value));
            }

            values =
                resolved;

            return true;
        }

        protected virtual double ResolveScalar(
            object? value)
        {
            return _scalar.Resolve(
                value);
        }
        protected double ResolveTopologyScalar<TResult>(
            INeighborSequenceOperator topology,
            INeighborScalarOperator<TResult> scalar,
            System.Func<TResult, double> conversion)
        {
            if (topology == null)
            {
                throw new ArgumentNullException(
                    nameof(topology));
            }

            if (scalar == null)
            {
                throw new ArgumentNullException(
                    nameof(scalar));
            }

            if (conversion == null)
            {
                throw new ArgumentNullException(
                    nameof(conversion));
            }

            var relationships =
                ApplyNeighborOperator(
                    topology);

            var result =
                scalar.Apply(
                    this,
                    relationships);

            var value =
                conversion(
                    result);

            if (double.IsNaN(value) ||
                double.IsInfinity(value))
            {
                throw new ArithmeticException(
                    "Topology scalar must be finite.");
            }

            return value;
        }
        
        protected double ComposeEmission(
            double emission,
            double topology,
            IEmissionComposition composition)
        {
            if (composition == null)
            {
                throw new ArgumentNullException(
                    nameof(composition));
            }

            return composition.Compose(
                emission,
                topology);
        }
        
        protected bool TryResolveEmission<TResult>(
            string propertyName,
            INeighborSequenceOperator topology,
            INeighborScalarOperator<TResult> topologyScalar,
            Func<TResult, double> conversion,
            IEmissionComposition composition,
            out double value)
        {
            value =
                0.0;

            if (topology == null)
            {
                throw new ArgumentNullException(
                    nameof(topology));
            }

            if (topologyScalar == null)
            {
                throw new ArgumentNullException(
                    nameof(topologyScalar));
            }

            if (conversion == null)
            {
                throw new ArgumentNullException(
                    nameof(conversion));
            }

            if (composition == null)
            {
                throw new ArgumentNullException(
                    nameof(composition));
            }

            if (!TryEmit(
                    propertyName,
                    out var emission))
            {
                return false;
            }

            var resolvedTopology =
                ResolveTopologyScalar(
                    topology,
                    topologyScalar,
                    conversion);

            value =
                ComposeEmission(
                    emission,
                    resolvedTopology,
                    composition);

            return true;
        }
    }
}
