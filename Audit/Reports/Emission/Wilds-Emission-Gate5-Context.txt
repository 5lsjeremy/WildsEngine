    [Fact]
    public void ExponentialPropagation_AttenuatesIntensity_ByNeighborDistance()
    {
        TestLog.Section(
            "NEIGHBOR PROPAGATION - EXPONENTIAL DISTANCE ATTENUATION");

        TestLog.Test(
            "Proving propagation intensity attenuates according to Neighbor distance.");

        var topology =
            new NeighborTopologyFixture();

        const double sourceIntensity =
            1.0;

        const double decay =
            0.5;

        TestLog.Task(
            "Creating A -> B at distance 1.");

        Assert.True(
            topology.A.State.Process(
                new AddNeighborInstruction(
                    topology.B,
                    NeighborTopologyFixture.LocalMembership,
                    1.0)));

        TestLog.Task(
            "Creating A -> C at distance 3.");

        Assert.True(
            topology.A.State.Process(
                new AddNeighborInstruction(
                    topology.C,
                    NeighborTopologyFixture.LocalMembership,
                    3.0)));

        Assert.True(
            topology.A.Neighbors.TryGet(
                topology.B.Id,
                NeighborTopologyFixture.LocalMembership,
                out var relationshipB));

        Assert.True(
            topology.A.Neighbors.TryGet(
                topology.C.Id,
                NeighborTopologyFixture.LocalMembership,
                out var relationshipC));

        Assert.NotNull(
            relationshipB);

        Assert.NotNull(
            relationshipC);

        var propagation =
            new ExponentialDistancePropagation(
                decay);

        TestLog.State(
            $"Source intensity = {sourceIntensity}");

        TestLog.State(
            $"Decay = {decay}");

        TestLog.State(
            $"B distance = {relationshipB!.Distance}");

        TestLog.State(
            $"C distance = {relationshipC!.Distance}");

        TestLog.Task(
            "Propagating source intensity independently to B and C.");

        var intensityB =
            propagation.Propagate(
                sourceIntensity,
                relationshipB);

        var intensityC =
            propagation.Propagate(
                sourceIntensity,
                relationshipC);

        var expectedB =
            sourceIntensity *
            Math.Exp(
                -decay *
                relationshipB.Distance);

        var expectedC =
            sourceIntensity *
            Math.Exp(
                -decay *
                relationshipC.Distance);

        TestLog.Expected(
            $"B intensity = {expectedB:R}");

        TestLog.Expected(
            $"C intensity = {expectedC:R}");

        TestLog.State(
            $"B received intensity = {intensityB:R}");

        TestLog.State(
            $"C received intensity = {intensityC:R}");

        Assert.Equal(
            expectedB,
            intensityB,
            12);

        Assert.Equal(
            expectedC,
            intensityC,
            12);

        Assert.True(
            intensityB <
            sourceIntensity);

        Assert.True(
            intensityC <
            intensityB);

        Assert.True(
            intensityC >
            0.0);

        TestLog.Pass(
            "Propagation attenuated monotonically with increasing Neighbor distance.");

        TestLog.Pass(
            "Exponential Neighbor propagation certification completed successfully.");
    }

using System;
using WildsEngine.Core.Abstractions;
using WildsEngine.Core.Abstractions.Neighbors.Propagation;

namespace WildsEngine.Core.Neighbors.Propagation
{
    public sealed class ExponentialDistancePropagation :
        INeighborPropagation
    {
        private readonly double _decay;

        public ExponentialDistancePropagation(
            double decay)
        {
            ValidateNonNegativeFinite(
                decay,
                nameof(decay));

            _decay =
                decay;
        }

        public double Propagate(
            double intensity,
            INeighborRelationship relationship)
        {
            if (relationship == null)
            {
                throw new ArgumentNullException(
                    nameof(relationship));
            }

            ValidateNonNegativeFinite(
                intensity,
                nameof(intensity));

            var distance =
                relationship.Distance;

            ValidateNonNegativeFinite(
                distance,
                nameof(relationship));

            var exponent =
                -_decay *
                distance;

            /*
             * Very large finite decay and distance values can
             * overflow their product to negative infinity.
             *
             * For non-negative decay and distance, that limit
             * legitimately represents complete attenuation:
             *
             *     Exp(-Infinity) = 0
             *
             * NaN remains invalid and must never pass through.
             */
            if (double.IsNaN(
                    exponent))
            {
                throw new ArithmeticException(
                    "Propagation produced an invalid exponent.");
            }

            var attenuation =
                Math.Exp(
                    exponent);

            if (double.IsNaN(
                    attenuation) ||
                double.IsInfinity(
                    attenuation))
            {
                throw new ArithmeticException(
                    "Propagation produced non-finite attenuation.");
            }

            var propagated =
                intensity *
                attenuation;

            if (double.IsNaN(
                    propagated) ||
                double.IsInfinity(
                    propagated))
            {
                throw new ArithmeticException(
                    "Propagation produced a non-finite result.");
            }

            return propagated;
        }

        private static void ValidateNonNegativeFinite(
            double value,
            string parameterName)
        {
            if (double.IsNaN(
                    value) ||
                double.IsInfinity(
                    value) ||
                value < 0.0)
            {
                throw new ArgumentOutOfRangeException(
                    parameterName);
            }
        }
    }
}

using System;
using WildsEngine.Core.Abstractions.Neighbors.Propagation;

namespace WildsEngine.Core.Neighbors.Propagation
{
    public sealed class MinimumIntensityEligibility :
        INeighborPropagationEligibility
    {
        public MinimumIntensityEligibility(
            double minimumIntensity)
        {
            ValidateIntensity(
                minimumIntensity,
                nameof(minimumIntensity));

            MinimumIntensity =
                minimumIntensity;
        }

        public double MinimumIntensity { get; }

        public bool IsEligible(
            double intensity)
        {
            ValidateIntensity(
                intensity,
                nameof(intensity));

            return intensity >=
                   MinimumIntensity;
        }

        private static void ValidateIntensity(
            double value,
            string parameterName)
        {
            if (double.IsNaN(
                    value) ||
                double.IsInfinity(
                    value) ||
                value < 0.0)
            {
                throw new ArgumentOutOfRangeException(
                    parameterName);
            }
        }
    }
}

using System;
using System.Collections.Generic;
using WildsEngine.Core.Abstractions;
namespace WildsEngine.Core.Topology
{
    public sealed class NeighborPropagation<TResult> : INeighborPropagation<TResult>
    {
        private readonly INeighborPropagationRule<TResult> _rule;
        public NeighborPropagation(INeighborPropagationRule<TResult> rule){_rule=rule??throw new ArgumentNullException(nameof(rule));}
        public IReadOnlyList<TResult> Propagate(IBaseObject origin,string membership,double maximumDistance)
        {
            if(origin==null)throw new ArgumentNullException(nameof(origin));if(string.IsNullOrWhiteSpace(membership))throw new ArgumentException("Membership cannot be empty.",nameof(membership));membership=membership.Trim();NeighborDistance.Validate(maximumDistance,nameof(maximumDistance));
            var results=new List<TResult>();var best=new Dictionary<Guid,double>();var queue=new List<Entry>();best[origin.Id]=0d;queue.Add(new Entry(origin,0d));while(queue.Count>0){var bi=0;for(var i=1;i<queue.Count;i++)if(queue[i].Distance<queue[bi].Distance)bi=i;var e=queue[bi];queue.RemoveAt(bi);var rels=e.Object.Neighbors.Relationships;for(var i=0;i<rels.Count;i++){var r=rels[i];if(!string.Equals(r.Membership,membership,StringComparison.Ordinal))continue;var next=e.Distance+r.Distance;if(next>maximumDistance)continue;double old;if(best.TryGetValue(r.Neighbor.Id,out old)&&old<=next)continue;best[r.Neighbor.Id]=next;results.Add(_rule.Apply(origin,e.Object,r,next));queue.Add(new Entry(r.Neighbor,next));}}return results.ToArray();
        }
        private sealed class Entry{public Entry(IBaseObject o,double d){Object=o;Distance=d;}public IBaseObject Object{get;}public double Distance{get;}}
    }
}
