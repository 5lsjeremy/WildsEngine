  380:         private static async Task
  381:             ConcurrentLoadRespectsLimit()
  382:         {
  383:             const int limit = 3;
  384:             const int count = 24;
  385: 
  386:             using var engine =
  387:                 CreateEngine(
  388:                     "certification.concurrent",
  389:                     retainedInstructions: limit,
  390:                     concurrentInstructions: limit);
  391: 
  392:             Assert(
  393:                 engine.Initialize(),
  394:                 "Initialize() should succeed.");
  395: 
  396:             var current = 0;
  397:             var maximum = 0;
  398: 
  399:             var release =
  400:                 new TaskCompletionSource<bool>(
  401:                     TaskCreationOptions
  402:                         .RunContinuationsAsynchronously);
  403: 
  404:             var entered =
  405:                 new TaskCompletionSource<bool>(
  406:                     TaskCreationOptions
  407:                         .RunContinuationsAsynchronously);
  408: 
  409:             var tasks =
  410:                 new List<Task<int>>(
  411:                     count);
  412: 
  413:             for (var i = 0; i < count; i++)
  414:             {
  415:                 tasks.Add(
  416:                     engine.ExecuteInstructionAsync<int>(
  417:                             async (_, token) =>
  418:                             {
  419:                                 var now =
  420:                                     Interlocked.Increment(
  421:                                         ref current);
  422: 
  423:                                 UpdateMaximum(
  424:                                     ref maximum,
  425:                                     now);
  426: 
  427:                                 if (now >= limit)
  428:                                 {
  429:                                     entered.TrySetResult(
  430:                                         true);
  431:                                 }
  432: 
  433:                                 try
  434:                                 {
  435:                                     await release.Task
  436:                                         .ConfigureAwait(false);
  437: 
  438:                                     token.ThrowIfCancellationRequested();
  439: 
  440:                                     return 1;
  441:                                 }
  442:                                 finally
  443:                                 {
  444:                                     Interlocked.Decrement(
  445:                                         ref current);
  446:                                 }
  447:                             })
  448:                         .AsTask());
  449:             }
  450: 
  451:             await entered.Task
  452:                 .ConfigureAwait(false);
  453: 
  454:             LogPool(
  455:                 engine,
  456:                 "At concurrency limit");
  457: 
  458:             Assert(
  459:                 engine.Instructions.ActiveCount <= limit,
  460:                 "Active leases exceeded MaximumConcurrent.");
  461: 
  462:             release.TrySetResult(
  463:                 true);
  464: 
  465:             await Task.WhenAll(
  466:                     tasks)
  467:                 .ConfigureAwait(false);
  468: 
  469:             Assert(
  470:                 maximum == limit,
  471:                 "Concurrent load did not exercise the configured limit.");
  472: 
  473:             Assert(
  474:                 engine.Instructions.ActiveCount == 0,
  475:                 "All concurrent leases should be returned.");
  476: 
  477:             Assert(
  478:                 engine.Instructions.CreatedCount <= limit,
  479:                 "Pool created more processors than " +
  480:                 "concurrent leasing permits.");
  481:         }

  483:         private static void UpdateMaximum(
  484:             ref int target,
  485:             int value)
  486:         {
  487:             while (true)
  488:             {
  489:                 var current =
  490:                     Volatile.Read(
  491:                         ref target);

  493:                 if (value <= current)
  494:                     return;
  495: 
  496:                 if (Interlocked.CompareExchange(
  497:                         ref target,
  498:                         value,
  499:                         current) == current)
  500:                 {
  501:                     return;
  502:                 }
  503:             }
  504:         }

    6:     /// <summary>
    7:     /// Represents temporary ownership of an object rented from a pool.
    8:     /// Disposing the lease returns the object to its originating pool.
    9:     /// </summary>
   10:     public sealed class ObjectLease<T> : IDisposable
   11:         where T : class
   12:     {
   13:         private IObjectPool<T>? _pool;
   14:         private T? _instance;
   15: 
   16:         internal ObjectLease(
   17:             IObjectPool<T> pool,
   18:             T instance)
   19:         {
   20:             _pool = pool ??
   21:                     throw new ArgumentNullException(nameof(pool));
   22: 
   23:             _instance = instance ??
   24:                         throw new ArgumentNullException(nameof(instance));
   25:         }
   26: 
   27:         public T Instance
   28:         {
   29:             get
   30:             {
   31:                 return _instance ??
   32:                        throw new ObjectDisposedException(
   33:                            nameof(ObjectLease<T>));
   34:             }
   35:         }
   36: 
   37:         public void Dispose()
   38:         {
   39:             var instance =
   40:                 Interlocked.Exchange(
   41:                     ref _instance,
   42:                     null);
   43: 
   44:             var pool =
   45:                 Interlocked.Exchange(
   46:                     ref _pool,
   47:                     null);

    1: using System;
    2: using System.Collections.Concurrent;
    3: using System.Threading;
    4: using System.Threading.Tasks;
    5: using WildsEngine.Core.Abstractions;
    6: 
    7: namespace WildsEngine.Core.Pooling
    8: {
    9:     /// <summary>
   10:     /// Thread-safe object pool with bounded retention
   11:     /// and bounded concurrent leasing.
   12:     /// </summary>
   13:     public class ObjectPool<T> :
   14:         IObjectPool<T>,
   15:         IDisposable
   16:         where T : class
   17:     {
   18:         private readonly ConcurrentQueue<T> _available =
   19:             new ConcurrentQueue<T>();
   20: 
   21:         private readonly Func<T> _factory;
   22: 
   23:         private readonly SemaphoreSlim _concurrencyGate;
   24: 
   25:         private int _availableCount;
   26:         private int _createdCount;
   27:         private int _activeCount;
   28:         private int _disposed;
   29: 
   30:         public ObjectPool(
   31:             Func<T> factory,
   32:             int maximumRetained,
   33:             int maximumConcurrent)
   34:         {
   35:             _factory = factory ??
   36:                        throw new ArgumentNullException(nameof(factory));
   37: 
   38:             if (maximumRetained < 0)
   39:             {
   40:                 throw new ArgumentOutOfRangeException(
   41:                     nameof(maximumRetained));
   42:             }
   43: 
   44:             if (maximumConcurrent <= 0)
   45:             {
   46:                 throw new ArgumentOutOfRangeException(
   47:                     nameof(maximumConcurrent));
   48:             }
   49: 
   50:             MaximumRetained = maximumRetained;
   51:             MaximumConcurrent = maximumConcurrent;
   52: 
   53:             _concurrencyGate = new SemaphoreSlim(
   54:                 maximumConcurrent,
   55:                 maximumConcurrent);
   56:         }
   57: 
   58:         public int AvailableCount =>
   59:             Volatile.Read(ref _availableCount);
   60: 
   61:         public int CreatedCount =>
   62:             Volatile.Read(ref _createdCount);
   63: 
   64:         public int ActiveCount =>
   65:             Volatile.Read(ref _activeCount);
   66: 
   67:         public int MaximumRetained { get; }
   68: 
   69:         public int MaximumConcurrent { get; }
   70: 
   71:         public ObjectLease<T> Rent()
   72:         {
   73:             ThrowIfDisposed();
   74: 
   75:             _concurrencyGate.Wait();
   76: 
   77:             try
   78:             {
   79:                 // The pool may have been disposed while waiting.
   80:                 ThrowIfDisposed();
   81: 
   82:                 return CreateLease();
   83:             }
   84:             catch
   85:             {
   86:                 _concurrencyGate.Release();
   87: 
   88:                 throw;
   89:             }
   90:         }
   91: 
   92:         public async ValueTask<ObjectLease<T>> RentAsync(
   93:             CancellationToken cancellationToken = default)
   94:         {
   95:             ThrowIfDisposed();
   96: 
   97:             await _concurrencyGate
   98:                 .WaitAsync(cancellationToken)
   99:                 .ConfigureAwait(false);
  100: 
  101:             try
  102:             {
  103:                 // The pool may have been disposed while waiting.
  104:                 ThrowIfDisposed();
  105: 
  106:                 return CreateLease();
  107:             }
  108:             catch
  109:             {
  110:                 _concurrencyGate.Release();
  111: 
  112:                 throw;
  113:             }
  114:         }
  115: 
  116:         public bool Return(T instance)
  117:         {
  118:             if (instance == null)
  119:                 throw new ArgumentNullException(nameof(instance));
  120: 
  121:             var remainingActive =
  122:                 Interlocked.Decrement(ref _activeCount);
  123: 
  124:             if (remainingActive < 0)
  125:             {
  126:                 Interlocked.Increment(ref _activeCount);
  127: 
  128:                 throw new InvalidOperationException(
  129:                     "An object was returned without an active lease.");
  130:             }
  131: 
  132:             try
  133:             {
  134:                 if (Volatile.Read(ref _disposed) != 0)
  135:                 {
  136:                     DisposeInstance(instance);
  137: 
  138:                     return false;
  139:                 }
  140: 
  141:                 try
  142:                 {
  143:                     if (instance is
  144:                         IResettablePoolObject
  145:                         resettable)
  146:                     {
  147:                         if (!resettable.TryReset())
  148:                         {
  149:                             DisposeInstance(instance);
  150: 
  151:                             return false;
  152:                         }
  153:                     }
  154:                 }
  155:                 catch
  156:                 {
  157:                     DisposeInstance(instance);
  158: 
  159:                     throw;
  160:                 }
  161: 
  162:                 var retained =
  163:                     TryRetain(instance);
  164: 
  165:                 if (!retained)
  166:                     DisposeInstance(instance);
  167: 
  168:                 return retained;
  169:             }
  170:             finally
  171:             {
  172:                 _concurrencyGate.Release();
  173:             }
  174:         }
  175: 
  176:         public void Dispose()
  177:         {
  178:             if (Interlocked.Exchange(
  179:                     ref _disposed,
  180:                     1) != 0)
  181:             {
  182:                 return;
  183:             }
  184: 
  185:             while (_available.TryDequeue(out var instance))
  186:             {
  187:                 Interlocked.Decrement(
  188:                     ref _availableCount);
  189: 
  190:                 DisposeInstance(instance);
  191:             }
  192: 
  193:             // Intentionally do not dispose the semaphore here.
  194:             // Active leases and waiting renters may still need
  195:             // to interact with it while observing pool disposal.
  196:         }
  197: 
  198:         private ObjectLease<T>
  199:             CreateLease()
  200:         {
  201:             T instance;
  202: 
  203:             if (_available.TryDequeue(out var pooled))
  204:             {
  205:                 Interlocked.Decrement(
  206:                     ref _availableCount);
  207: 
  208:                 instance = pooled;
  209:             }
  210:             else
  211:             {
  212:                 instance = _factory();
  213: 
  214:                 if (instance == null)
  215:                 {
  216:                     throw new InvalidOperationException(
  217:                         "The object pool factory returned null.");
  218:                 }
  219: 
  220:                 Interlocked.Increment(
  221:                     ref _createdCount);
  222:             }
  223: 
  224:             Interlocked.Increment(
  225:                 ref _activeCount);
  226: 
  227:             return new ObjectLease<T>(
  228:                 this,
  229:                 instance);
  230:         }
  231: 
  232:         private bool TryRetain(T instance)
  233:         {
  234:             while (true)
  235:             {
  236:                 if (Volatile.Read(ref _disposed) != 0)
  237:                     return false;
  238: 
  239:                 var available =
  240:                     Volatile.Read(ref _availableCount);
  241: 
  242:                 if (available >= MaximumRetained)
  243:                     return false;
  244: 
  245:                 if (Interlocked.CompareExchange(
  246:                         ref _availableCount,
  247:                         available + 1,
  248:                         available) != available)
  249:                 {
  250:                     continue;
  251:                 }
  252: 
  253:                 if (Volatile.Read(ref _disposed) != 0)
  254:                 {
  255:                     Interlocked.Decrement(
  256:                         ref _availableCount);
  257: 
  258:                     return false;
  259:                 }
  260: 
  261:                 _available.Enqueue(instance);
  262: 
  263:                 return true;
  264:             }
  265:         }
  266: 
  267:         private static void DisposeInstance(
  268:             T instance)
  269:         {
  270:             if (instance is IDisposable disposable)
  271:                 disposable.Dispose();
  272:         }
  273: 
  274:         private void ThrowIfDisposed()
  275:         {
  276:             if (Volatile.Read(ref _disposed) != 0)
  277:             {
  278:                 throw new ObjectDisposedException(
  279:                     nameof(ObjectPool<T>));
  280:             }
  281:         }
  282:     }
  283: }

   16:         private readonly ITaskRegistry _registry;
   17: 
   18:         public ProcessorPoolingBenchmark(
   19:             IComponent component,
   20:             ITaskRegistry registry)
   21:         {
   22:             _component = component ??
   23:                 throw new ArgumentNullException(
   24:                     nameof(component));
   25: 
   26:             _registry = registry ??
   27:                 throw new ArgumentNullException(
   28:                     nameof(registry));
   29:         }
   30: 
   31:         public ProcessorPoolingBenchmarkResult Run(
   32:             int iterations,
   33:             int maximumRetained = 64,
   34:             int maximumConcurrent = 64)
   35:         {
   36:             if (iterations <= 0)
   37:             {
   38:                 throw new ArgumentOutOfRangeException(
   39:                     nameof(iterations));
   40:             }
   41: 
   42:             if (maximumRetained < 0)
   43:             {
   44:                 throw new ArgumentOutOfRangeException(
   45:                     nameof(maximumRetained));
   46:             }
   47: 
   48:             if (maximumConcurrent <= 0)
   49:             {
   50:                 throw new ArgumentOutOfRangeException(
   51:                     nameof(maximumConcurrent));
   52:             }

   55: 
   56:             var allocationMeasurementAvailable =
   57:                 IsAllocationMeasurementAvailable();
   58: 
   59:             var directAllocatedBytesBefore =
   60:                 GetAllocatedBytes();
   61: 
   62:             var directWatch =
   63:                 Stopwatch.StartNew();
   64: 
   65:             for (var i = 0; i < iterations; i++)
   66:             {
   67:                 var processor =
   68:                     new TaskProcessor(
   69:                         _component,
   70:                         "benchmark.direct",
   71:                         _registry);
   72: 
   73:                 Consume(
   74:                     processor);
   75:             }
   76: 
   77:             directWatch.Stop();
   78: 
   79:             var directAllocatedBytes =
   80:                 allocationMeasurementAvailable
   81:                     ? GetAllocatedBytes() -
   82:                       directAllocatedBytesBefore
   83:                     : 0;

   86:                 new ObjectPool<ITaskProcessor>(
   87:                     factory: () =>
   88:                         new TaskProcessor(
   89:                             _component,
   90:                             "benchmark.pooled",
   91:                             _registry),
   92:                     maximumRetained:
   93:                         maximumRetained,
   94:                     maximumConcurrent:
   95:                         maximumConcurrent))

   98:                     GetAllocatedBytes();

  100:                 var pooledWatch =
  101:                     Stopwatch.StartNew();

  103:                 for (var i = 0; i < iterations; i++)
  104:                 {
  105:                     using (var lease = pool.Rent())
  106:                     {
  107:                         Consume(
  108:                             lease.Instance);
  109:                     }
  110:                 }
  111: 
  112:                 pooledWatch.Stop();
  113: 
  114:                 var pooledAllocatedBytes =
  115:                     allocationMeasurementAvailable
  116:                         ? GetAllocatedBytes() -
  117:                           pooledAllocatedBytesBefore
  118:                         : 0;

  121:                     iterations,
  122:                     directWatch.Elapsed,
  123:                     pooledWatch.Elapsed,

  127:                     allocationMeasurementAvailable);
  128:             }
  129:         }

  139:                         _registry);

  141:                 Consume(
  142:                     processor);
  143:             }
  144:         }
