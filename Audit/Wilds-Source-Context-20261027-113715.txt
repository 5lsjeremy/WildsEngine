        [Fact]
        public void State_processes_external_reference_type_without_core_knowledge()
        {
            WildsLog.Info(
                "Starting external reference-type processing test.");

            var component = new TestComponent("test");

            var payload = new ConsumerThing(
                "External Consumer Object",
                42);

            WildsLog.Process(
                $"Created consumer-owned payload. " +
                $"Type={payload.GetType().FullName}, " +
                $"Name={payload.Name}, " +
                $"Value={payload.Value}.");

            var instruction = new SetContextVariableInstruction(
                "consumer",
                payload);

            WildsLog.Process(
                $"Processing instruction. " +
                $"Id={instruction.Id}, " +
                $"Intent={instruction.Intent}, " +
                $"Key={instruction.Key}.");

            var processed =
                component.State.Process(instruction);

            WildsLog.State(
                $"Instruction processed. " +
                $"Result={processed}, " +
                $"VariableExists=" +
                $"{component.State.Context.Variables.ContainsKey("consumer")}.");

            Assert.True(processed);
            Assert.True(
                component.State.Context.Variables.ContainsKey(
                    "consumer"));

            var stored =
                component.State.Context.Variables["consumer"];

            WildsLog.State(
                $"Retrieved stored value. " +
                $"Type={stored?.GetType().FullName ?? "<null>"}.");

            Assert.Same(payload, stored);

            var typed =
                Assert.IsType<ConsumerThing>(stored);

            Assert.Equal(
                "External Consumer Object",
                typed.Name);

            Assert.Equal(42, typed.Value);

            WildsLog.Info(
                "External reference-type processing test passed.");
        }

    public float Z { get; }

    public bool Equals(ConsumerVector other)
    {
        return X.Equals(other.X)
               && Y.Equals(other.Y)
               && Z.Equals(other.Z);
    }

    public override bool Equals(object? obj)
    {
        return obj is ConsumerVector other
               && Equals(other);

    {
        unchecked

            var hash = 17;

            hash = (hash * 31) + X.GetHashCode();
            hash = (hash * 31) + Y.GetHashCode();

        {
            var values = new Dictionary<string, object?>
            {
                ["target"] = null
            };

            IReadOnlyDictionary<string, object?> readOnly = values;

            Assert.True(readOnly.ContainsKey("target"));
            Assert.False(readOnly.ContainsKey("missing"));

        {
            var expected = new ConsumerThing("external");

            var values = new Dictionary<string, object?>
            {
                ["target"] = expected
            };

            values["target"] = null;

            Assert.True(values.ContainsKey("target"));
            Assert.Null(values["target"]);
        }

        {
            var expected = new ConsumerThing("external");

            var values = new Dictionary<string, object?>
            {
                ["target"] = null

using WildsEngine.Core.Abstractions;
using WildsEngine.Core.Objects;
using WildsEngine.Core.Processing;

namespace WildsEngine.Core.Tests;

public sealed class TestState :
    BaseObject,
    IState
{
    private readonly IComponent _component;

    public TestState(
        IComponent component,
        IContext context,
        string layer)
        : base(layer)
    {
        _component = component
                     ?? throw new ArgumentNullException(
                         nameof(component));

        Context = context
                  ?? throw new ArgumentNullException(
                      nameof(context));
    }

    public override IComponent Component =>
        _component;

    public IContext Context { get; }

    public bool Process(
        IStateInstruction instruction)
    {
        if (instruction == null)
        {
            throw new ArgumentNullException(
                nameof(instruction));
        }

        if (instruction is not ISetContextVariableInstruction setVariable)
        {
            return false;
        }

        Context.SetVariable(
            setVariable.Key,
            setVariable.Value);

        return true;
    }
}

    public abstract class WildsContext :

    {
        private readonly Dictionary<string, object?> _variables =
            new Dictionary<string, object?>();

        private readonly Dictionary<string, object?> _contextMetadata =
            new Dictionary<string, object?>();

        private readonly IComponent _component;

        protected WildsContext(
            IComponent component,
            string layer)
            : base(layer)
        {
            _component = component
                ?? throw new ArgumentNullException(
                    nameof(component));
        }

        protected WildsContext(
            Guid id,
            IComponent component,
            string layer)
            : base(id, layer)
        {
            _component = component
                ?? throw new ArgumentNullException(
                    nameof(component));
        }

        public sealed override IComponent Component =>
            _component;

        public IReadOnlyDictionary<string, object?> Variables =>
            _variables;

        IReadOnlyDictionary<string, object?> IContext.Metadata =>
            _contextMetadata;

        public virtual void SetVariable(
            string key,
            object? value)
        {
            ValidateKey(key);

            _variables[key] = value;
        }

        public virtual void SetMetadata(
            string key,
            object? value)
        {
            ValidateKey(key);

            _contextMetadata[key] = value;
        }

        protected bool RemoveVariable(string key)
        {
            ValidateKey(key);

            return _variables.Remove(key);
        }

        protected bool RemoveMetadata(string key)
        {
            ValidateKey(key);

            return _contextMetadata.Remove(key);
        }

        protected bool TryGetVariable(
            string key,
            out object? value)
        {
            ValidateKey(key);

            return _variables.TryGetValue(
                key,
                out value);
        }

        protected bool TryGetMetadata(
            string key,
            out object? value)
        {
            ValidateKey(key);

            return _contextMetadata.TryGetValue(
                key,
                out value);
        }

        protected static void ValidateKey(string key)
        {
            if (string.IsNullOrWhiteSpace(key))
                throw new ArgumentException(
                    "Context keys cannot be empty.",
                    nameof(key));
        }
    }

using System;

namespace WildsEngine.Core.Diagnostics
{
    public static class WildsLog
    {
        public static Action<string>? Writer { get; set; }

        public static bool Enabled { get; set; } = true;

        public static void Info(string message)
        {
            Write("INFO", message);
        }

        public static void Process(string message)
        {
            Write("PROCESS", message);
        }

        public static void State(string message)
        {
            Write("STATE", message);
        }

        public static void Warning(string message)
        {
            Write("WARN", message);
        }

        public static void Error(string message)
        {
            Write("ERROR", message);
        }

        private static void Write(
            string category,
            string message)
        {
            if (!Enabled)
                return;

            var formatted =
                $"[Wilds:{category}] {message}";

            if (Writer != null)
            {
                Writer(formatted);
                return;
            }

            Console.WriteLine(formatted);
        }
    }
}

using System;
using System.Collections.Generic;
using WildsEngine.Core.Abstractions;

namespace WildsEngine.Core.Objects
{
    public abstract class BaseObject : IBaseObject
    {
        private readonly List<IBaseObject> _neighbors =
            new List<IBaseObject>();

        protected BaseObject(string layer)
            : this(Guid.NewGuid(), layer)
        {
        }

        protected BaseObject(Guid id, string layer)
        {
            if (id == Guid.Empty)
                throw new ArgumentException(
                    "Wilds objects require a non-empty GUID.",
                    nameof(id));

            if (string.IsNullOrWhiteSpace(layer))
                throw new ArgumentException(
                    "Wilds objects require a layer.",
                    nameof(layer));

            Id = id;
            Slug = BuildSlug(id, layer);
        }

        public Guid Id { get; }

        public string Slug { get; private set; }

        public IReadOnlyList<IBaseObject> Neighbors =>
            _neighbors.AsReadOnly();

        public abstract IComponent Component { get; }

        // Component shortcuts.
        // These intentionally do NOT belong to IBaseObject.

        public virtual IState State =>
            Component.State;

        public virtual ISignal Signal =>
            Component.Signal;

        public virtual IDictionary<string, object?> Metadata =>
            Component.Metadata;

        public string AppendSlug(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return Slug;

            var segment = value.Trim().Trim('.');

            if (segment.Length == 0)
                return Slug;

            Slug = $"{Slug}.{segment}";

            return Slug;
        }

        public bool AddNeighbor(IBaseObject neighbor)
        {
            if (neighbor == null)
                throw new ArgumentNullException(nameof(neighbor));

            if (neighbor.Id == Id)
                return false;

            for (var i = 0; i < _neighbors.Count; i++)
            {
                if (_neighbors[i].Id == neighbor.Id)
                    return false;
            }

            _neighbors.Add(neighbor);

            return true;
        }

        public bool RemoveNeighbor(Guid id)
        {
            for (var i = 0; i < _neighbors.Count; i++)
            {
                if (_neighbors[i].Id != id)
                    continue;

                _neighbors.RemoveAt(i);

                return true;
            }

            return false;
        }

        private static string BuildSlug(Guid id, string layer)
        {
            var raw = id.ToString("N");

            var first = raw.Substring(0, 3);
            var last = raw.Substring(raw.Length - 3, 3);

            return $"{first}{last}-{layer.Trim()}";
        }
    }
}
